create or replace function public.client_proposal_signature_guard()
returns trigger
language plpgsql
set search_path = ''
as $$
begin
  if tg_op = 'UPDATE' then
    -- The FK's own ON DELETE SET NULL (template_id) arrives from inside the
    -- referential-action trigger; nothing else may change.
    if pg_trigger_depth() > 1
       and (to_jsonb(new) - 'template_id') = (to_jsonb(old) - 'template_id')
       and new.template_id is null then
      return new;
    end if;
    raise exception 'loe_signature_append_only: a signature record is never changed'
      using errcode = '23514';
  end if;
  -- INSERT: the server clock, whatever the caller sent.
  new.signed_at := now();
  return new;
end
$$;
