-- ── 1. client_proposal_signing_words: the attorney's own words ───────────────────────────────────────────────────────
-- One row per version per firm. status: draft -> approved -> retired. At most one draft and one approved per firm.
-- Edits (draft only) by the admin tier; approval by a named approver (client_proposal_setting.approver_user_ids) whose role can
-- write proposals. Approving retires the previous approved version. An approved or retired row is frozen.
-- Merge fields: link_body must contain {{signing_link}} exactly once; the four email fields may use {{client_name}},
-- {{firm_name}} and {{link_expires}} (link email only); the four page fields use none. Nothing else in double braces.
-- v3 (evaluator B5, non-blocking): EVERY check constraint in this file is written `check (coalesce(<rule>, false))`. A CHECK
-- that evaluates to NULL passes, so a rule over a nullable column (or a regex over one) would otherwise let a NULL through.
-- The functions supply values today; the coalesce makes the tables refuse a NULL even if a future function does not.
create table if not exists public.client_proposal_signing_words (
  id             uuid        primary key default gen_random_uuid(),
  org_id         uuid        not null references public.crm_org(id) on delete cascade,
  version        integer     not null,
  status         text        not null default 'draft',
  -- Page words (client-facing, the attorney's): the line on the consent button, the full consent text, the line beside Sign,
  -- the line under the signature box.
  consent_line   text        not null,
  consent_text   text        not null,
  sign_line      text        not null,
  box_note       text        not null,
  -- Email words: the signing-link email and the signed-copy email (Sprint B sends the second).
  link_subject   text        not null,
  link_body      text        not null,
  copy_subject   text        not null,
  copy_body      text        not null,
  created_by     uuid        null default auth.uid() references auth.users(id) on delete set null,
  updated_by     uuid        null references auth.users(id) on delete set null,
  approved_by    uuid        null references auth.users(id) on delete set null,
  approved_at    timestamptz null,
  created_at     timestamptz not null default now(),
  updated_at     timestamptz not null default now(),
  constraint client_proposal_signing_words_status check (coalesce(status in ('draft', 'approved', 'retired'), false)),
  constraint client_proposal_signing_words_version_pos check (coalesce(version >= 1, false)),
  constraint client_proposal_signing_words_org_version unique (org_id, version),
  constraint client_proposal_signing_words_lens check (coalesce(
        char_length(btrim(consent_line)) between 1 and 300
    and char_length(btrim(consent_text)) between 1 and 8000
    and char_length(btrim(sign_line))    between 1 and 600
    and char_length(btrim(box_note))     between 1 and 600
    and char_length(btrim(link_subject)) between 1 and 200
    and char_length(btrim(link_body))    between 1 and 8000
    and char_length(btrim(copy_subject)) between 1 and 200
    and char_length(btrim(copy_body))    between 1 and 8000
  , false)),
  -- Subjects are one line.
  constraint client_proposal_signing_words_subject_line check (coalesce(link_subject !~ '[\r\n]' and copy_subject !~ '[\r\n]', false)),
  constraint client_proposal_signing_words_approved_stamp check (coalesce(status = 'draft' or (approved_by is not null and approved_at is not null), false))
);

create unique index if not exists client_proposal_signing_words_one_draft
  on public.client_proposal_signing_words (org_id) where status = 'draft';
create unique index if not exists client_proposal_signing_words_one_approved
  on public.client_proposal_signing_words (org_id) where status = 'approved';

comment on table public.client_proposal_signing_words is
  'The firm attorney''s own words for the emailed signing link (0117): consent line and text, the line beside Sign, the line under the signature box, and both emails. Versioned; approved by a named approver; approved rows frozen. Lectual writes none of these words.';

-- The merge-field rule, shared by the approve check below and mirrored by src/lib/loe/signing-words.ts.
create or replace function public.client_proposal_signing_words_fields_ok(
  p_text text, p_allowed text[], p_required text
) returns boolean
language plpgsql
immutable
set search_path = ''
as $$
declare
  v_field text;
  v_count integer := 0;
begin
  if p_text is null then return false; end if;
  -- An unclosed "{{" or a stray "}}" outside a field.
  if pg_catalog.regexp_replace(p_text, '\{\{[a-z_]+\}\}', '', 'g') ~ '(\{\{|\}\})' then return false; end if;
  for v_field in select (pg_catalog.regexp_matches(p_text, '\{\{([a-z_]+)\}\}', 'g'))[1] loop
    if not (v_field = any (p_allowed)) then return false; end if;
    if p_required is not null and v_field = p_required then v_count := v_count + 1; end if;
  end loop;
  return p_required is null or v_count = 1;
end
$$;
revoke all on function public.client_proposal_signing_words_fields_ok(text, text[], text) from public, anon, authenticated;

create or replace function public.client_proposal_signing_words_rules()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_role     text := public.current_org_role();
  v_uid      uuid := auth.uid();
  v_approver boolean;
begin
  -- The service role (no session) is never a writer of the firm's words.
  if v_uid is null then
    raise exception 'signing_words_not_allowed: the firm''s signing words are written by its own staff' using errcode = '42501';
  end if;

  if tg_op = 'INSERT' then
    if v_role is null or v_role not in ('owner', 'admin', 'senior_admin') then
      raise exception 'signing_words_edit_admin_only: only the admin tier starts a draft' using errcode = '42501';
    end if;
    new.status := 'draft';
    new.created_by := v_uid;
    new.updated_by := v_uid;
    new.approved_by := null;
    new.approved_at := null;
    new.created_at := now();
    new.updated_at := now();
    -- Serialise version numbers per firm.
    perform pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtext('client_proposal_signing_words:' || new.org_id::text));
    select coalesce(max(w.version), 0) + 1 into new.version
      from public.client_proposal_signing_words w where w.org_id = new.org_id;
    return new;
  end if;

  -- UPDATE
  if new.org_id is distinct from old.org_id or new.version is distinct from old.version
     or new.created_by is distinct from old.created_by or new.created_at is distinct from old.created_at then
    raise exception 'signing_words_frozen: those columns never change' using errcode = '23514';
  end if;

  if old.status = 'draft' and new.status = 'draft' then
    if v_role is null or v_role not in ('owner', 'admin', 'senior_admin') then
      raise exception 'signing_words_edit_admin_only: only the admin tier edits a draft' using errcode = '42501';
    end if;
    new.updated_by := v_uid;
    new.updated_at := now();
    new.approved_by := null;
    new.approved_at := null;
    return new;
  end if;

  if old.status = 'draft' and new.status = 'approved' then
    -- Approve is its own act: no word may change in the same statement.
    if (to_jsonb(new) - array['status','approved_by','approved_at','updated_at','updated_by'])
       is distinct from (to_jsonb(old) - array['status','approved_by','approved_at','updated_at','updated_by']) then
      raise exception 'signing_words_bad_transition: save the words first, then approve them on their own' using errcode = '23514';
    end if;
    select (v_uid = any (s.approver_user_ids)) into v_approver
      from public.client_proposal_setting s where s.org_id = old.org_id;
    if v_approver is not true
       or v_role is null
       or v_role not in ('owner','admin','senior_admin','intake','paralegal','law_clerk','attorney','clerk') then
      raise exception 'signing_words_not_approver: only a named approver approves the signing words' using errcode = '42501';
    end if;
    if not public.client_proposal_signing_words_fields_ok(new.link_body, array['signing_link','client_name','firm_name','link_expires'], 'signing_link')
       or not public.client_proposal_signing_words_fields_ok(new.link_subject, array['client_name','firm_name'], null)
       or not public.client_proposal_signing_words_fields_ok(new.copy_subject, array['client_name','firm_name'], null)
       or not public.client_proposal_signing_words_fields_ok(new.copy_body, array['client_name','firm_name'], null)
       or not public.client_proposal_signing_words_fields_ok(new.consent_line, array[]::text[], null)
       or not public.client_proposal_signing_words_fields_ok(new.consent_text, array[]::text[], null)
       or not public.client_proposal_signing_words_fields_ok(new.sign_line, array[]::text[], null)
       or not public.client_proposal_signing_words_fields_ok(new.box_note, array[]::text[], null) then
      raise exception 'signing_words_fields_invalid: a merge field is unknown, unclosed, or the link field is missing' using errcode = '23514';
    end if;
    -- Retire the version it replaces (the partial unique index allows one approved per firm).
    update public.client_proposal_signing_words w
       set status = 'retired', updated_at = now()
     where w.org_id = old.org_id and w.status = 'approved' and w.id <> old.id;
    new.approved_by := v_uid;
    new.approved_at := now();
    new.updated_by := v_uid;
    new.updated_at := now();
    return new;
  end if;

  -- The one other transition: approved -> retired, done above by the approve of a newer version (depth > 1).
  if old.status = 'approved' and new.status = 'retired' and pg_catalog.pg_trigger_depth() > 1
     and (to_jsonb(new) - array['status','updated_at']) = (to_jsonb(old) - array['status','updated_at']) then
    new.updated_at := now();
    return new;
  end if;

  raise exception 'signing_words_frozen: approved and retired signing words never change' using errcode = '23514';
end
$$;
revoke all on function public.client_proposal_signing_words_rules() from public, anon, authenticated;

do $$
begin
  if not exists (select 1 from pg_trigger where tgname = 'client_proposal_signing_words_rules_trg'
      and tgrelid = 'public.client_proposal_signing_words'::regclass) then
    create trigger client_proposal_signing_words_rules_trg
      before insert or update on public.client_proposal_signing_words
      for each row execute function public.client_proposal_signing_words_rules();
  end if;
end
$$;

alter table public.client_proposal_signing_words enable row level security;
revoke all on public.client_proposal_signing_words from anon;
revoke all on public.client_proposal_signing_words from authenticated;
grant select on public.client_proposal_signing_words to authenticated;

  on public.client_proposal_signing_words to authenticated;
grant update (status, consent_line, consent_text, sign_line, box_note, link_subject, link_body, copy_subject, copy_body)
  on public.client_proposal_signing_words to authenticated;

do $$
begin
  if not exists (select 1 from pg_policies where schemaname = 'public' and tablename = 'client_proposal_signing_words'
      and policyname = 'client_proposal_signing_words_select') then
    -- INTENDED (evaluator N12): every member of the firm, viewer and social_media included, reads the firm's own words (the
    -- spec: "Every member ... reads"), like 0106's letter templates. The request and event policies below are narrower on
    -- purpose (they hold client evidence); the words hold none.
    create policy client_proposal_signing_words_select on public.client_proposal_signing_words
      for select using (org_id = public.current_org_id());
  end if;
  if not exists (select 1 from pg_policies where schemaname = 'public' and tablename = 'client_proposal_signing_words'
      and policyname = 'client_proposal_signing_words_insert') then
    create policy client_proposal_signing_words_insert on public.client_proposal_signing_words
      for insert with check (
        org_id = public.current_org_id()
        and public.current_org_role() in ('owner', 'admin', 'senior_admin')
      );
  end if;
  if not exists (select 1 from pg_policies where schemaname = 'public' and tablename = 'client_proposal_signing_words'
      and policyname = 'client_proposal_signing_words_update') then
    -- The admin tier (edit) or a named approver (approve); the trigger decides which act each may do.
    create policy client_proposal_signing_words_update on public.client_proposal_signing_words
      for update using (
        org_id = public.current_org_id()
        and (
          public.current_org_role() in ('owner', 'admin', 'senior_admin')
          or exists (select 1 from public.client_proposal_setting s
                      where s.org_id = public.current_org_id() and auth.uid() = any (s.approver_user_ids))
        )
      )
      with check (org_id = public.current_org_id());
  end if;
end
$$;

-- ── 2. client_proposal_signing_request: one emailed link for one proposal version ───────────────────────────────────
create table if not exists public.client_proposal_signing_request (
  id                    uuid        primary key default gen_random_uuid(),
  org_id                uuid        not null references public.crm_org(id) on delete cascade,
  proposal_id           uuid        not null,
  family_id             uuid        not null,
  version               integer     not null,
  -- sha256 (hex) of the url-safe token (>= 32 random bytes). The token itself is never stored.
  token_sha256          text        not null unique,
  status                text        not null default 'sending',
  closed_reason         text        null,
  -- Where the link went: the PROPOSAL ROW's client_email at opening (never an address from the browser).
  sent_to               text        not null,
  sent_at               timestamptz null,
  expires_at            timestamptz not null,
  -- Frozen at opening. The letter is final here: the emailed PDF, the signing page and the signed PDF come from THIS text.
  signer_name           text        not null,
  signer_title          text        null,
  entity_name           text        null,
  address               jsonb       null,
  details               jsonb       not null,
  template_version      integer     null,
  agreement_text        text        not null,
  agreement_sha256      text        not null,
  canonical_fingerprint text        not null,
  review_pdf_sha256     text        null,
  selection             jsonb       not null,
  snapshot              jsonb       not null,
  date_text             text        not null,
  -- The approved signing words this request uses, snapshotted (text + sha256), so a later approval never changes an open link.
  -- No FK: words rows are never deleted except with the whole firm, and the snapshot below is what counts.
  words_id              uuid        not null,
  words_version         integer     not null,
  consent_line          text        not null,
  consent_text          text        not null,
  consent_sha256        text        not null,
  sign_line             text        not null,
  box_note              text        not null,
  copy_subject          text        not null,
  copy_body             text        not null,
  words_sha256          text        not null,
  opened_at             timestamptz null,
  consent_at            timestamptz null,
  signed_at             timestamptz null,
  -- ON DELETE CASCADE, not SET NULL: a signature row is removed only by the owner's family delete (0102), which removes this
  -- request too; a SET NULL would be an update the guard refuses and would break the `signed` check.
  signature_id          uuid        null references public.client_proposal_signature(id) on delete cascade,
  created_at            timestamptz not null default now(),
  updated_at            timestamptz not null default now(),
  constraint client_proposal_signing_request_proposal_org_fk foreign key (proposal_id, org_id)
    references public.client_proposal (id, org_id) on delete cascade,
  constraint client_proposal_signing_request_status check (coalesce(status in ('sending', 'open', 'signed', 'closed'), false)),
  constraint client_proposal_signing_request_closed_reason check (coalesce(
    (status = 'closed') = (closed_reason is not null)
    and (closed_reason is null or closed_reason in (
      'resent', 'send_failed', 'expired', 'withdrawn', 'superseded', 'accepted_elsewhere', 'proposal_changed', 'reset'
    ))
  , false)),
  constraint client_proposal_signing_request_sha check (coalesce(
        token_sha256 ~ '^[0-9a-f]{64}$' and agreement_sha256 ~ '^[0-9a-f]{64}$' and canonical_fingerprint ~ '^[0-9a-f]{64}$'
    and consent_sha256 ~ '^[0-9a-f]{64}$' and words_sha256 ~ '^[0-9a-f]{64}$'
    and (review_pdf_sha256 is null or review_pdf_sha256 ~ '^[0-9a-f]{64}$')
  , false)),
  constraint client_proposal_signing_request_text_len check (coalesce(char_length(agreement_text) between 1 and 204800, false)),
  constraint client_proposal_signing_request_name_len check (coalesce(char_length(btrim(signer_name)) between 2 and 200, false)),
  constraint client_proposal_signing_request_email check (coalesce(char_length(sent_to) between 3 and 320 and sent_to ~ '^[^@\s]+@[^@\s]+$', false)),
  constraint client_proposal_signing_request_expiry check (coalesce(expires_at > created_at and expires_at <= created_at + interval '7 days 1 minute', false)),
  constraint client_proposal_signing_request_signed check (coalesce((status = 'signed') = (signature_id is not null and signed_at is not null), false)),
  constraint client_proposal_signing_request_shapes check (coalesce(
    jsonb_typeof(details) = 'object' and pg_column_size(details) <= 8192
    and (address is null or (jsonb_typeof(address) = 'object' and pg_column_size(address) <= 4096))
    and jsonb_typeof(snapshot) = 'object'
  , false))
);

-- One request in flight per proposal version (sending or open).
create unique index if not exists client_proposal_signing_request_one_open
  on public.client_proposal_signing_request (proposal_id) where status in ('sending', 'open');
create index if not exists client_proposal_signing_request_org_idx
  on public.client_proposal_signing_request (org_id, proposal_id, created_at desc);

comment on table public.client_proposal_signing_request is
  'One emailed signing link for one proposal version (0117). Written only by the service role through the signing_* functions. Staff read status, timestamps and ids; the token hash, the letter text, the address, the details, the email, the snapshot and the words are withheld. Privileged client content: never logged.';

-- Status changes and evidence columns move only through the functions below; a direct service-role update is limited to
-- review_pdf_sha256 (set once, by the send action, before the email goes out).
create or replace function public.client_proposal_signing_request_guard()
returns trigger
language plpgsql
set search_path = ''
as $$
begin
  if tg_op = 'UPDATE' then
    if coalesce(current_setting('lectual.signing_fn', true), '') <> new.id::text then
      if (to_jsonb(new) - array['review_pdf_sha256','updated_at']) is distinct from (to_jsonb(old) - array['review_pdf_sha256','updated_at'])
         or old.review_pdf_sha256 is not null or old.status <> 'sending' then
        raise exception 'signing_request_guarded: a signing request changes only through its functions' using errcode = '42501';
      end if;
    end if;
    new.updated_at := now();
    return new;
  end if;
  -- INSERT only through signing_open.
  if coalesce(current_setting('lectual.signing_fn', true), '') <> 'open' then
    raise exception 'signing_request_guarded: a signing request is opened only through client_proposal_signing_open' using errcode = '42501';
  end if;
  new.created_at := now();
  new.updated_at := now();
  return new;
end
$$;
revoke all on function public.client_proposal_signing_request_guard() from public, anon, authenticated;

-- ── 3. client_proposal_signing_event: append-only evidence ───────────────────────────────────────────────────────────
create table if not exists public.client_proposal_signing_event (
  id                uuid        primary key default gen_random_uuid(),
  org_id            uuid        not null references public.crm_org(id) on delete cascade,
  request_id        uuid        not null references public.client_proposal_signing_request(id) on delete cascade,
  kind              text        not null,
  at                timestamptz not null default now(),
  ip                inet        null,
  user_agent        text        null,
  -- sent
  sent_to           text        null,
  -- consent
  consent_sha256    text        null,
  -- signature
  signature_kind    text        null,
  signature_png     bytea       null,
  typed_style       text        null,
  signature_sha256  text        null,
  typed_name        text        null,
  -- signed
  agreement_sha256  text        null,
  -- closed
  closed_reason     text        null,
  constraint client_proposal_signing_event_kind check (coalesce(kind in ('sent', 'opened', 'consent', 'signature', 'signed', 'closed'), false)),
  constraint client_proposal_signing_event_ua_len check (coalesce(user_agent is null or char_length(user_agent) <= 400, false)),
  -- v3 (evaluator B5): wrapped in coalesce(..., false). A CHECK that evaluates to NULL PASSES, so without it a `consent` row
  -- with a NULL consent_sha256 (NULL ~ regex = NULL), or a `signature` row with a NULL signature_sha256 or typed_style, passed.
  constraint client_proposal_signing_event_shape check (coalesce(
    case kind
      when 'sent'      then sent_to is not null
      when 'opened'    then true
      when 'consent'   then consent_sha256 ~ '^[0-9a-f]{64}$'
      when 'signature' then typed_name is not null and char_length(btrim(typed_name)) between 2 and 200
                            and signature_sha256 ~ '^[0-9a-f]{64}$'
                            and (
                              (signature_kind = 'drawn' and signature_png is not null and typed_style is null
                                and octet_length(signature_png) between 8 and 102400
                                and substring(signature_png from 1 for 8) = '\x89504e470d0a1a0a'::bytea)
                              or (signature_kind = 'typed' and signature_png is null and typed_style in ('script', 'print'))
                            )
      when 'signed'    then agreement_sha256 ~ '^[0-9a-f]{64}$'
      when 'closed'    then closed_reason is not null
    end, false)
  )
);
create index if not exists client_proposal_signing_event_request_idx on public.client_proposal_signing_event (request_id, at);

comment on table public.client_proposal_signing_event is

Written only inside the signing_* functions; the time is the database clock.';

-- ── 5. client_proposal_signing_open (service role) ───────────────────────────────────────────────────────────────────
-- Locks the proposal, checks it is the same live version with a letter, checks the words are the firm's APPROVED version and
-- that the snapshotted words and text hash to what is passed, closes any request already in flight for this version
-- (reason `resent`), and inserts the new one as `sending`. org_id, family_id, version and sent_to come from the proposal row.
-- Returns the new request id. SECURITY DEFINER: it never changes the proposal.
create or replace function public.client_proposal_signing_open(
  p_proposal_id           uuid,
  p_org_id                uuid,
  p_family_id             uuid,
  p_version               integer,
  p_token_sha256          text,
  p_expires_at            timestamptz,
  p_signer_name           text,
  p_signer_title          text,
  p_entity_name           text,
  p_address               jsonb,
  p_details               jsonb,
  p_template_version      integer,
  p_agreement_text        text,
  p_agreement_sha256      text,
  p_canonical_fingerprint text,
  p_selection             jsonb,
  p_snapshot              jsonb,
  p_date_text             text,
  p_words_id              uuid,
  p_words_sha256          text
)
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_row   public.client_proposal%rowtype;
  v_words public.client_proposal_signing_words%rowtype;
  v_old   uuid;
  v_id    uuid;
begin
  select * into v_row from public.client_proposal p where p.id = p_proposal_id for update;
  if not found
     or v_row.org_id is distinct from p_org_id
     or v_row.family_id is distinct from p_family_id
     or v_row.version is distinct from p_version
     or v_row.status is distinct from 'sent'
     or (v_row.expires_on is not null and v_row.expires_on < (now() at time zone 'Etc/GMT+12')::date)
     or (jsonb_typeof(v_row.content -> 'loe') = 'object') is not true then
    raise exception 'signing_changed: the proposal is not the live version with a letter' using errcode = '23514';
  end if;
  if v_row.client_email is null or btrim(v_row.client_email) !~ '^[^@\s]+@[^@\s]+$' then
    raise exception 'signing_no_email: the proposal has no client email on file' using errcode = '23514';
  end if;
  if not public.client_proposal_signing_letter_current(v_row.org_id, v_row.content) then
    raise exception 'signing_letter_replaced: the firm has replaced this letter since the proposal went out' using errcode = '23514';
  end if;

  select * into v_words from public.client_proposal_signing_words w where w.id = p_words_id;
  if not found or v_words.org_id is distinct from v_row.org_id or v_words.status is distinct from 'approved' then
    raise exception 'signing_words_changed: those are not the firm''s approved signing words' using errcode = '23514';
  end if;
  -- The words hash: sha256 of the eight fields joined by U+001F, in this order (src/lib/loe/signing-words.ts wordsSha256).
  if pg_catalog.encode(pg_catalog.sha256(pg_catalog.convert_to(
       v_words.consent_line || chr(31) || v_words.consent_text || chr(31) || v_words.sign_line || chr(31) || v_words.box_note
       || chr(31) || v_words.link_subject || chr(31) || v_words.link_body || chr(31) || v_words.copy_subject || chr(31) || v_words.copy_body,
       'UTF8')), 'hex') is distinct from p_words_sha256 then
    raise exception 'signing_words_changed: the words changed while the link was being made' using errcode = '23514';
  end if;

  if p_token_sha256 is null or p_token_sha256 !~ '^[0-9a-f]{64}$'
     or p_agreement_text is null or p_agreement_sha256 is null
     or pg_catalog.encode(pg_catalog.sha256(pg_catalog.convert_to(p_agreement_text, 'UTF8')), 'hex') is distinct from p_agreement_sha256 then
    raise exception 'signing_invalid: the agreement text does not match its fingerprint' using errcode = '23514';
  end if;
  -- The app passes (its now + 7 days - 5 minutes), clamped to the proposal's own last day (v2, N2), so a clock skew of up to
  -- 5 minutes between the app and the database never refuses a send.
  if p_expires_at is null or p_expires_at <= now() or p_expires_at > now() + interval '7 days' then
    raise exception 'signing_invalid: a link lasts at most 7 days' using errcode = '23514';
  end if;

  -- Pressing the button again sends a fresh link and closes the old one. A request already past its expiry is closed as
  -- `expired`, not `resent`, so the evidence says why it closed (v2, N5). Proposal locked above, requests now (lock order).
  for v_old in select r.id from public.client_proposal_signing_request r
                where r.proposal_id = v_row.id and r.status in ('sending', 'open') and r.expires_at <= now()
                order by r.id for update loop
    perform public.client_proposal_signing_close_internal(v_old, 'expired');
  end loop;
  for v_old in select r.id from public.client_proposal_signing_request r
                where r.proposal_id = v_row.id and r.status in ('sending', 'open')
                order by r.id for update loop
    perform public.client_proposal_signing_close_internal(v_old, 'resent');
  end loop;

  perform set_config('lectual.signing_fn', 'open', true);
  insert into public.client_proposal_signing_request (
    org_id, proposal_id, family_id, version, token_sha256, status, sent_to, expires_at,
    signer_name, signer_title, entity_name, address, details, template_version,
    agreement_text, agreement_sha256, canonical_fingerprint, selection, snapshot, date_text,
    words_id, words_version, consent_line, consent_text, consent_sha256, sign_line, box_note, copy_subject, copy_body, words_sha256
  ) values (
    v_row.org_id, v_row.id, v_row.family_id, v_row.version, p_token_sha256, 'sending', btrim(v_row.client_email), p_expires_at,
    btrim(p_signer_name), nullif(btrim(coalesce(p_signer_title, '')), ''), nullif(btrim(coalesce(p_entity_name, '')), ''),
    p_address, p_details, p_template_version,
    p_agreement_text, p_agreement_sha256, p_canonical_fingerprint, p_selection, p_snapshot, p_date_text,
    v_words.id, v_words.version, v_words.consent_line, v_words.consent_text,
    pg_catalog.encode(pg_catalog.sha256(pg_catalog.convert_to(v_words.consent_text, 'UTF8')), 'hex'),
    v_words.sign_line, v_words.box_note, v_words.copy_subject, v_words.copy_body, p_words_sha256
  )
  returning id into v_id;
  perform set_config('lectual.signing_fn', '', true);
  return v_id;
end
$$;
revoke all on function public.client_proposal_signing_open(
  uuid, uuid, uuid, integer, text, timestamptz, text, text, text, jsonb, jsonb, integer, text, text, text, jsonb, jsonb, text, uuid, text)
  from public, anon, authenticated;
grant execute on function public.client_proposal_signing_open(

  to service_role;

-- ── 6. Sent, opened, consent, close (service role) ───────────────────────────────────────────────────────────────────
-- After the email: p_sent true -> `open` + a `sent` event; false -> `closed` (send_failed): a failed send leaves no open request.
create or replace function public.client_proposal_signing_mark_sent(p_request_id uuid, p_sent boolean)

language plpgsql
security definer
set search_path = ''
as $$

-- The link is opened. The FIRST open is recorded (opened_at + one `opened` event with ip and browser). Returns the request's
-- state for the page: 'open' / 'signed' / 'closed' / 'expired' / 'unknown'. An open request past expires_at is closed here
-- (reason `expired`). Looked up by token sha256 only (the app hashes the token; the hash is never compared in app code with ==).
create or replace function public.client_proposal_signing_opened(p_token_sha256 text, p_ip inet, p_user_agent text)
returns text
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_req public.client_proposal_signing_request%rowtype;
begin
  if p_token_sha256 is null or p_token_sha256 !~ '^[0-9a-f]{64}$' then return 'unknown'; end if;
  select * into v_req from public.client_proposal_signing_request r where r.token_sha256 = p_token_sha256 for update;
  if not found then return 'unknown'; end if;
  if v_req.status = 'signed' then return 'signed'; end if;
  if v_req.status <> 'open' then return 'closed'; end if;
  if v_req.expires_at <= now() then
    perform public.client_proposal_signing_close_internal(v_req.id, 'expired');
    return 'expired';
  end if;
  if v_req.opened_at is null then
    perform set_config('lectual.signing_fn', v_req.id::text, true);
    update public.client_proposal_signing_request r set opened_at = now() where r.id = v_req.id;
    insert into public.client_proposal_signing_event (org_id, request_id, kind, ip, user_agent)
    values (v_req.org_id, v_req.id, 'opened', p_ip, left(p_user_agent, 400));
    perform set_config('lectual.signing_fn', '', true);
  end if;
  return 'open';
end
$$;

-- ── 7. Closing a link when the version leaves `sent` (withdrawn, superseded, accepted in-app, reset) ─────────────────
-- AFTER UPDATE OF status on client_proposal. client_proposal_signing_sign marks its request `signed` BEFORE it updates the
-- proposal, so its own acceptance finds nothing to close. A reset (accepted -> sent) closes nothing that is still open (a
-- signed request is not in flight); any request still in flight on a status change is closed with a reason for that change.
create or replace function public.client_proposal_signing_on_status()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_id uuid;
  v_reason text;
begin
  if new.status is not distinct from old.status then return null; end if;
  v_reason := case new.status
    when 'withdrawn'  then 'withdrawn'
    when 'superseded' then 'superseded'
    when 'accepted'   then 'accepted_elsewhere'
    else 'proposal_changed'
  end;
  -- The proposal row is already locked by the UPDATE that fired this trigger; the requests are locked second (lock order).
  for v_id in select r.id from public.client_proposal_signing_request r
               where r.proposal_id = new.id and r.status in ('sending', 'open')
               order by r.id for update loop
    perform public.client_proposal_signing_close_internal(
      v_id,
      case when (select r2.expires_at from public.client_proposal_signing_request r2 where r2.id = v_id) <= now()
           then 'expired' else v_reason end);
  end loop;
  return null;
end
$$;
revoke all on function public.client_proposal_signing_on_status() from public, anon, authenticated;

do $$
begin
  if not exists (select 1 from pg_trigger where tgname = 'client_proposal_signing_on_status_trg'
      and tgrelid = 'public.client_proposal'::regclass) then
    create trigger client_proposal_signing_on_status_trg
      after update of status on public.client_proposal
      for each row execute function public.client_proposal_signing_on_status();
  end if;
end
$$;

-- ── 8. client_proposal_signing_sign: THE one signing function (service role) ─────────────────────────────────────────
-- Locks the proposal, then the request (by token sha256; v2 lock order), checks the request is open, unexpired, opened and
-- consented, checks the typed name is the frozen signer name (ASCII case and spaces folded), checks the signature payload,
-- checks the proposal is still the same live version with the firm's CURRENT letter, writes 0106's signature row from the request's FROZEN text, appends the `signature` and `signed`
-- evidence, marks the request `signed`, and accepts the proposal with the request's frozen selection and snapshot, all in
-- this one transaction. A second call for a signed request returns (signature_id, false) and changes nothing.
-- SECURITY INVOKER, service_role only: 0106's lifecycle accepts a letter proposal only from the service role with no session
-- and a signature row from this very transaction (the reason 0106's sign_and_accept and 0113's adobe_complete are INVOKER).
-- NO EXCEPTION BLOCK, NO SAVEPOINT (the lifecycle looks for a signature row with this transaction's own xid).
create or replace function public.client_proposal_signing_sign(
  p_token_sha256     text,
  p_typed_name       text,
  p_signature_kind   text,
  p_signature_png    bytea,
  p_typed_style      text,
  p_signature_sha256 text,
  p_ip               inet,
  p_user_agent       text
)
returns table (signature_id uuid, signed_now boolean, proposal_id uuid, signed_at timestamptz)
language plpgsql
security invoker
set search_path = ''
as $$
declare
  v_req          public.client_proposal_signing_request%rowtype;
  v_row          public.client_proposal%rowtype;
  v_signature_id uuid := gen_random_uuid();
  v_template_id  uuid;
  v_template_ver integer;
  v_linked_ver   integer;
  v_loe          jsonb;
  v_signed_at    timestamptz;
  v_fold_typed   text;
  v_fold_frozen  text;
  v_proposal_id  uuid;
begin
  -- Lock order (v2, N1): the PROPOSAL first, then the request. Learn the proposal id without a lock, lock the proposal, then
  -- lock the request and re-check everything below on the LOCKED row.
  select r.proposal_id into v_proposal_id from public.client_proposal_signing_request r where r.token_sha256 = p_token_sha256;
  if not found then
    raise exception 'signing_link_closed: no such link' using errcode = '23514';
  end if;
  select * into v_row from public.client_proposal p where p.id = v_proposal_id for update;
  select * into v_req from public.client_proposal_signing_request r where r.token_sha256 = p_token_sha256 for update;
  if not found or v_req.proposal_id is distinct from v_proposal_id then
    raise exception 'signing_link_closed: no such link' using errcode = '23514';
  end if;
  if v_req.status = 'signed' then
    return query select v_req.signature_id, false, v_req.proposal_id, v_req.signed_at;
    return;
  end if;
  if v_req.status <> 'open' or v_req.expires_at <= now() then
    raise exception 'signing_link_closed: this link is no longer open' using errcode = '23514';
  end if;
  if v_req.opened_at is null or v_req.consent_at is null then
    raise exception 'signing_no_consent: consent comes before signing' using errcode = '23514';
  end if;

  -- Name folding (v2, N3): trim, runs of whitespace to one space, and ASCII A-Z to a-z ONLY (translate, not lower(), whose
  -- result depends on the database collation). Non-ASCII letters compare exactly. The app folds by the same rule
  -- (src/lib/loe/email-signing/name-fold.ts), and passes p_typed_name already cleaned by cleanTypedName.
  v_fold_typed  := pg_catalog.translate(pg_catalog.regexp_replace(pg_catalog.btrim(coalesce(p_typed_name, '')), '\s+', ' ', 'g'),
                     'ABCDEFGHIJKLMNOPQRSTUVWXYZ', 'abcdefghijklmnopqrstuvwxyz');
  v_fold_frozen := pg_catalog.translate(pg_catalog.regexp_replace(pg_catalog.btrim(v_req.signer_name), '\s+', ' ', 'g'),
                     'ABCDEFGHIJKLMNOPQRSTUVWXYZ', 'abcdefghijklmnopqrstuvwxyz');
  if char_length(v_fold_typed) < 2 or v_fold_typed is distinct from v_fold_frozen then
    raise exception 'signing_name_mismatch: the typed name is not the name on the letter' using errcode = '23514';
  end if;
  if p_signature_sha256 is null or p_signature_sha256 !~ '^[0-9a-f]{64}$' then
    raise exception 'signing_bad_signature: no signature' using errcode = '23514';
  end if;
  -- A NULL kind matches neither branch and is refused by the `else` (pinned by isolation case 29).
  if p_signature_kind = 'drawn' then
    if p_signature_png is null or octet_length(p_signature_png) > 102400
       or substring(p_signature_png from 1 for 8) is distinct from '\x89504e470d0a1a0a'::bytea
       or pg_catalog.encode(pg_catalog.sha256(p_signature_png), 'hex') is distinct from p_signature_sha256 then
      raise exception 'signing_bad_signature: the drawn signature is not a PNG of at most 100 KB, or its hash differs' using errcode = '23514';
    end if;
  elsif p_signature_kind = 'typed' then
    -- v3 (evaluator B5): every comparison is NULL-safe. `p_typed_style not in (...)` and `x <> y` are NULL when an input is
    -- NULL, and `NULL or NULL` would not raise, so a NULL style with any 64-hex hash would pass unverified. Hence the explicit
    -- `is null` and `is distinct from`.
    if p_signature_png is not null or p_typed_style is null or p_typed_style not in ('script', 'print')
       or p_typed_name is null
       -- p_typed_name is the app's cleanTypedName output (v2, N3), so btrim() is the identity here and both sides hash the
       -- same string: sha256(style || U+001F || cleaned name). Sprint A offers only 'print' (ruling N9); 'script' is reserved.
       or pg_catalog.encode(pg_catalog.sha256(pg_catalog.convert_to(p_typed_style || chr(31) || pg_catalog.btrim(p_typed_name), 'UTF8')), 'hex') is distinct from p_signature_sha256 then
      raise exception 'signing_bad_signature: the typed signature does not match its hash' using errcode = '23514';
    end if;
  else
    raise exception 'signing_bad_signature: drawn or typed' using errcode = '23514';
  end if;

  -- v_row was locked at the top (lock order); checked here, after the request's own checks.
  if v_row.id is null
     or v_row.org_id is distinct from v_req.org_id
     or v_row.family_id is distinct from v_req.family_id
     or v_row.version is distinct from v_req.version
     or v_row.status is distinct from 'sent'
     or (v_row.expires_on is not null and v_row.expires_on < (now() at time zone 'Etc/GMT+12')::date)
     or (jsonb_typeof(v_row.content -> 'loe') = 'object') is not true then
    raise exception 'signing_changed: the proposal changed after the link was sent' using errcode = '23514';
  end if;
  -- A link never signs a letter the firm has replaced since (v2, ruling B3). The app closes the link (`proposal_changed`)
  -- when it sees this; here it is only refused, because a refusal inside this function rolls back the whole transaction.
  if not public.client_proposal_signing_letter_current(v_row.org_id, v_row.content) then
    raise exception 'signing_letter_replaced: the firm has replaced this letter since the link was sent' using errcode = '23514';
  end if;
  if pg_catalog.encode(pg_catalog.sha256(pg_catalog.convert_to(v_req.agreement_text, 'UTF8')), 'hex') is distinct from v_req.agreement_sha256 then
    raise exception 'signing_text_mismatch: the stored text does not hash to its fingerprint' using errcode = '23514';
  end if;

  -- 0106's template link rule, copied: only the proposal's firm's approved (or since retired) version the content names.
  v_loe := v_row.content -> 'loe';
  if (v_loe ->> 'template_id') ~ '^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$' then
    v_template_id := (v_loe ->> 'template_id')::uuid;
  end if;
  if (v_loe ->> 'template_version') ~ '^[0-9]{1,9}$' then
    v_template_ver := (v_loe ->> 'template_version')::integer;
  end if;
  if v_template_id is not null then
    select t.version into v_linked_ver from public.client_proposal_loe_template t
     where t.id = v_template_id and t.org_id = v_row.org_id and t.status in ('approved', 'retired') and t.version = v_template_ver;
    if found then v_template_ver := v_linked_ver; else v_template_id := null; end if;
  end if;

  -- 0106's signature row, from the request's FROZEN columns. signed_at is 0106's guard's own clock.
  insert into public.client_proposal_signature (
    id, org_id, proposal_id, family_id, version, signer_name, signer_title, entity_name,
    address, signer_email, ip, user_agent, agreement_text, agreement_sha256, template_id, template_version
  ) values (
    v_signature_id, v_row.org_id, v_row.id, v_row.family_id, v_row.version, v_req.signer_name, v_req.signer_title,
    v_req.entity_name, v_req.address, v_req.sent_to, p_ip, left(p_user_agent, 400), v_req.agreement_text,
    v_req.agreement_sha256, v_template_id, v_template_ver
  )
  returning signed_at into v_signed_at;

  -- Evidence, then the request is marked signed BEFORE the proposal moves (so section 7's trigger finds nothing to close).
  perform set_config('lectual.signing_fn', v_req.id::text, true);
  insert into public.client_proposal_signing_event (org_id, request_id, kind, ip, user_agent, signature_kind, signature_png, typed_style, signature_sha256, typed_name)
  values (v_req.org_id, v_req.id, 'signature', p_ip, left(p_user_agent, 400), p_signature_kind, p_signature_png,
          case when p_signature_kind = 'typed' then p_typed_style end, p_signature_sha256, btrim(p_typed_name));
  insert into public.client_proposal_signing_event (org_id, request_id, kind, ip, user_agent, agreement_sha256)
  values (v_req.org_id, v_req.id, 'signed', p_ip, left(p_user_agent, 400), v_req.agreement_sha256);
  update public.client_proposal_signing_request r
     set status = 'signed', signature_id = v_signature_id, signed_at = v_signed_at
   where r.id = v_req.id and r.status = 'open';
  perform set_config('lectual.signing_fn', '', true);

  -- 0106's marker and the one conditional update fenced on `status = 'sent'`.
  perform set_config('lectual.loe_signing', v_row.id::text, true);
  update public.client_proposal p
     set status = 'accepted',
         accepted_selection = v_req.selection,
         accepted_snapshot = v_req.snapshot || jsonb_build_object(
           'schema', 2,
           -- The snapshot was built at send time; the acceptance instant is the signing's (0106's clock).
           'accepted_at', v_signed_at,
           'agreement', jsonb_build_object(
             'signature_id', v_signature_id,
             'signer_name', v_req.signer_name,
             'signed_at', v_signed_at,
             'sha256', v_req.agreement_sha256,
             'template_version', v_template_ver,
             'method', 'email_link',
             'signing_request_id', v_req.id
           )
         )
   where p.id = v_row.id and p.org_id = v_row.org_id and p.status = 'sent';
  if not found then
    raise exception 'signing_changed: only a live proposal can be signed' using errcode = '23514';
  end if;
  perform set_config('lectual.loe_signing', '', true);

  return query select v_signature_id, true, v_row.id, v_signed_at;
end
$$;
revoke all on function public.client_proposal_signing_sign(text, text, text, bytea, text, text, inet, text) from public, anon, authenticated;
grant execute on function public.client_proposal_signing_sign(text, text, text, bytea, text, text, inet, text) to service_role;

-- ── 8b. Send failures and readiness (v3, evaluator B1: "a firm is never left with no way to sign") ──────────────────────
-- The app refuses in-app signing only while the emailed path is POSITIVELY ready for the proposal (sprint-a item 2 / 18).
-- Readiness asks the database the SAME preconditions signing_open checks (so the app and 0117 can never disagree about an
-- address, a replaced letter, the live version or the approved words), plus how many sends failed for this version in the
-- last 24 hours. A failure that happens BEFORE signing_open returned an id (signing_open refused or errored, its own 23505,
-- an rpc timeout) cannot close a request, so the app records it here instead. A failure AFTER the id exists closes the
-- request `send_failed` (mark_sent(false), else signing_close) and is counted from the request table; it is not ALSO
-- written here (no double count). Refusals caused by the client's own input (rate limit, missing details, nothing chosen,
-- an undescribed class) are refused by the app before any rpc and are never recorded: a holder of the bearer /p/ link
-- cannot trip the fallback and so skip the email proof.
create table if not exists public.client_proposal_signing_failure (
  id          uuid        primary key default gen_random_uuid(),
  org_id      uuid        not null references public.crm_org(id) on delete cascade,
  proposal_id uuid        not null,
  kind        text        not null,
  at          timestamptz not null default now(),
  constraint client_proposal_signing_failure_proposal_org_fk foreign key (proposal_id, org_id)
    references public.client_proposal (id, org_id) on delete cascade,
  constraint client_proposal_signing_failure_kind check (coalesce(kind in (
    'open_refused', 'open_error', 'mail_uncertain', 'close_error', 'other'
  ), false))
);
create index if not exists client_proposal_signing_failure_proposal_idx
  on public.client_proposal_signing_failure (proposal_id, at desc);
comment on table public.client_proposal_signing_failure is
  'Failed attempts to email a signing link that left no request to close (0117 v3). Kind and time only: no address, text, token or error text. Service role only; counted by client_proposal_signing_readiness.';
alter table public.client_proposal_signing_failure enable row level security;
revoke all on public.client_proposal_signing_failure from public, anon, authenticated;
grant select, insert on public.client_proposal_signing_failure to service_role;
-- No policy for authenticated: staff read nothing (kind and time only, nothing they act on). Append-only: no update or delete
-- grant, and the trigger below refuses both even for the service role.
-- A cascade (0102's owner family delete, or the firm's own removal) fires this row trigger with a trigger depth above 1, so
-- a DELETE is allowed only then, the same rule as the evidence table. A direct UPDATE or DELETE is refused, the service
-- role included.
create or replace function public.client_proposal_signing_failure_guard()
returns trigger
language plpgsql
set search_path = ''
as $$
begin
  if tg_op = 'INSERT' then
    new.at := now();
    return new;
  end if;
  if tg_op = 'DELETE' and pg_catalog.pg_trigger_depth() > 1 then
    return old;
  end if;
  raise exception 'signing_failure_append_only: failures are never changed or removed' using errcode = '42501';
end
$$;
revoke all on function public.client_proposal_signing_failure_guard() from public, anon, authenticated;
do $$
begin
  if not exists (select 1 from pg_trigger where tgname = 'client_proposal_signing_failure_guard_trg'
      and tgrelid = 'public.client_proposal_signing_failure'::regclass) then
    create trigger client_proposal_signing_failure_guard_trg
      before insert or update or delete on public.client_proposal_signing_failure
      for each row execute function public.client_proposal_signing_failure_guard();
  end if;
end
$$;

-- Record one failure for a proposal. Fenced on the proposal row's org; at most 50 rows per proposal per 24 hours (beyond
-- that the count is already far past the threshold). Returns whether a row was written. Never raises for a missing or
-- foreign proposal (returns false): the app calls it from failure paths and must not fail twice.
create or replace function public.client_proposal_signing_note_failure(p_proposal_id uuid, p_org_id uuid, p_kind text)
returns boolean
language plpgsql
security definer
set search_path = ''
as $$
begin
  if p_kind is null or p_kind not in ('open_refused', 'open_error', 'mail_uncertain', 'close_error', 'other') then
    return false;
  end if;
  if not exists (select 1 from public.client_proposal p where p.id = p_proposal_id and p.org_id = p_org_id) then
    return false;
  end if;
  if (select count(*) from public.client_proposal_signing_failure f
       where f.proposal_id = p_proposal_id and f.at > now() - interval '24 hours') >= 50 then
    return false;
  end if;
  insert into public.client_proposal_signing_failure (org_id, proposal_id, kind) values (p_org_id, p_proposal_id, p_kind);
  return true;
end
$$;
revoke all on function public.client_proposal_signing_note_failure(uuid, uuid, text) from public, anon, authenticated;
grant execute on function public.client_proposal_signing_note_failure(uuid, uuid, text) to service_role;

-- Readiness for ONE proposal version: signing_open's own preconditions, without writing, plus the failure count.
-- Returns jsonb { "state": "ok" | "changed" | "no_email" | "letter_replaced" | "no_words", "failures": <int> }.
-- failures = requests of THIS proposal row closed `send_failed` in the last 24 hours + failure rows in the last 24 hours.
-- The app treats anything but state "ok" with failures < 3, and any error, as NOT ready (the client signs in-app).
create or replace function public.client_proposal_signing_readiness(p_proposal_id uuid, p_org_id uuid)
returns jsonb
language plpgsql
stable
security definer
set search_path = ''
as $$
declare
  v_row      public.client_proposal%rowtype;
  v_state    text := 'ok';
  v_failures integer;
begin
  select * into v_row from public.client_proposal p where p.id = p_proposal_id;
  if not found
     or v_row.org_id is distinct from p_org_id
     or v_row.status is distinct from 'sent'
     or (v_row.expires_on is not null and v_row.expires_on < (now() at time zone 'Etc/GMT+12')::date)
     or (jsonb_typeof(v_row.content -> 'loe') = 'object') is not true then
    v_state := 'changed';
  elsif v_row.client_email is null or btrim(v_row.client_email) !~ '^[^@\s]+@[^@\s]+$' then
    v_state := 'no_email';
  elsif not public.client_proposal_signing_letter_current(v_row.org_id, v_row.content) then
    v_state := 'letter_replaced';
  elsif not exists (select 1 from public.client_proposal_signing_words w where w.org_id = v_row.org_id and w.status = 'approved') then
    v_state := 'no_words';
  end if;
  select (select count(*) from public.client_proposal_signing_request r
            where r.proposal_id = p_proposal_id and r.org_id = p_org_id and r.status = 'closed'
              and r.closed_reason = 'send_failed' and r.updated_at > now() - interval '24 hours')
       + (select count(*) from public.client_proposal_signing_failure f
            where f.proposal_id = p_proposal_id and f.org_id = p_org_id and f.at > now() - interval '24 hours')
    into v_failures;
  return jsonb_build_object('state', v_state, 'failures', coalesce(v_failures, 0));
end
$$;
revoke all on function public.client_proposal_signing_readiness(uuid, uuid) from public, anon, authenticated;
grant execute on function public.client_proposal_signing_readiness(uuid, uuid) to service_role;

-- 19. sign() succeeds once: one client_proposal_signature row (org_id = the proposal's), the proposal `accepted` with
--     accepted_snapshot.agreement.method = 'email_link', the request `signed`, two events (signature, signed); a second call
