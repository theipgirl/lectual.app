-- WHAT IT ADDS
--   lawmatics_connection.last_sync_at timestamptz null: the START time of the last COMPLETE prospect step of the Lawmatics daily
--   sync for that firm (both /prospects and /contacts read completely, includes kept, inside its budget, nobody left over the
--   per-run limit, no failed write). The sync's cutoff is max(2026-11-01, last_sync_at - 2 days); null = 2026-11-01.
--   No new table, so no new tenant table and no new isolation table: the 0077 cases cover the row. Two small cases are added
--   below for the column itself (a signed-in caller cannot write it; a member reads only their own firm's value).
--
-- WHO WRITES IT
--   Only the service-role runner (lectual.app src/lib/lawmatics/daily-sync.ts, `writeLastSync`), fenced on the connection's id AND
--   org_id. The firm's admin tier may update other columns of the row under 0077's policy, and 0077 grants insert / update / delete
--   to `authenticated` at TABLE level (0077 line 76), which a column-level revoke cannot narrow: Postgres checks the table grant
--   first. So the rule is a BEFORE INSERT OR UPDATE trigger, 0116's pattern: a caller that is not trusted may not set the column on
--   insert (it must stay null) and may not change it on update (OLD vs NEW, `is distinct from`). Trusted = `current_user` is
--   `service_role`, or the table's owner or a member of it (`pg_has_role(current_user, relowner, 'MEMBER')`; superusers are), or the
--   change arrives from inside a referential action (`pg_trigger_depth() > 1`). The function is SECURITY INVOKER on purpose, so
--   `current_user` is the caller's own role. Refusal: 42501 with the stable prefix `lawmatics_last_sync_service_only`. No grant
--   is changed for the other columns, so nothing 0077 allows today breaks (connect, reconnect, disconnect, the manual importer's
--   `recordLawmaticsOutcome`, the daily sync's `invalid` write).
--   lectual.app reads it with the scoped client for the Settings line "Last complete daily sync ..." (`getLawmaticsLastSync`).
--
-- WITHOUT IT (today)
--   The read answers 42703 (Postgres) / PGRST204 (PostgREST) and the app treats that as "not set up": the cutoff is 2026-11-01 on
--   every run, nothing is written, and Settings says "Showing the last daily sync isn't switched on for your firm yet. Lectual will
--   turn it on." Matching by lawmatics_id and email means a prospect seen before is never created twice. Known limit: a lead a
--   person DELETES comes back on the next run while its prospect is after the cutoff (always, until this is applied).
--

alter table public.lawmatics_connection
  add column if not exists last_sync_at timestamptz null;

comment on column public.lawmatics_connection.last_sync_at is
  'Start time of the last complete daily-sync prospect step (lectual.app src/lib/lawmatics/daily-sync.ts). Written only by the service role, fenced on id and org_id (trigger lawmatics_connection_last_sync_guard). Null = no complete run yet (the sync then plans from 2026-11-01).';

-- Read, like the other non-secret columns.
grant select (last_sync_at) on public.lawmatics_connection to authenticated;

-- Only a trusted role writes it. SECURITY INVOKER: current_user must be the caller's own role for the trust test.
create or replace function public.lawmatics_connection_last_sync_guard()
returns trigger
language plpgsql
set search_path = ''
as $$
declare
  v_owner oid;
begin
  -- A referential action (a firm delete cascading) is never a person setting the cutoff.
  if pg_trigger_depth() > 1 then
    return new;
  end if;
  if current_user = 'service_role' then
    return new;
  end if;
  select c.relowner into v_owner from pg_catalog.pg_class c where c.oid = tg_relid;
  if v_owner is not null and pg_catalog.pg_has_role(current_user, v_owner, 'MEMBER') then
    return new;
  end if;
  if tg_op = 'INSERT' then
    if new.last_sync_at is not null then
      raise exception 'lawmatics_last_sync_service_only: only the daily sync sets the last complete run'
        using errcode = '42501';
    end if;
  elsif new.last_sync_at is distinct from old.last_sync_at then
    raise exception 'lawmatics_last_sync_service_only: only the daily sync sets the last complete run'
      using errcode = '42501';
  end if;
  return new;
end;
$$;

revoke all on function public.lawmatics_connection_last_sync_guard() from public, anon, authenticated;

do $$
begin
  if not exists (
    select 1 from pg_catalog.pg_trigger
     where tgrelid = 'public.lawmatics_connection'::regclass
       and tgname = 'lawmatics_connection_last_sync_guard'
       and not tgisinternal
  ) then
    create trigger lawmatics_connection_last_sync_guard
      before insert or update on public.lawmatics_connection
      for each row execute function public.lawmatics_connection_last_sync_guard();
  end if;
end;
$$;

comment on function public.lawmatics_connection_last_sync_guard() is
  'lectual 0118: refuses (42501 lawmatics_last_sync_service_only) any signed-in insert that sets, or update that changes, lawmatics_connection.last_sync_at. Trusted: service_role, the table owner, referential actions.';
