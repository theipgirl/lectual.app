-- ── 1. adobe_sign_connection: a firm's OWN Adobe Acrobat Sign account ────────────────────────────────────────────────
create or replace function public.adobe_sign_host_ok(p_url text, p_path_allowed boolean)
returns boolean
language sql
immutable
set search_path = ''
as $$
  -- https://, no userinfo, no port, a host ending .adobesign.com or .echosign.com (with at least one label before it),
  -- and (for an access point) no path beyond "/". The app checks the same rule (src/lib/esign/host-rule.ts).
  select p_url is not null
     and p_url ~ '^https://[a-z0-9.-]+\.(adobesign|echosign)\.com(/[^?#[:space:]]*)?$'
     and p_url !~ '^https://[^/]*@'
     and p_url !~ '^https://(adobesign|echosign)\.com'
     and (p_path_allowed or p_url ~ '^https://[a-z0-9.-]+\.(adobesign|echosign)\.com/?$');
$$;
revoke all on function public.adobe_sign_host_ok(text, boolean) from public, anon, authenticated;

-- status -> revoked clears both secrets; every update touches updated_at.
create or replace function public.adobe_sign_connection_rules()
returns trigger
language plpgsql
set search_path = ''
as $$
begin
  if new.status = 'revoked' then
    new.access_token_enc := null;
    new.refresh_token_enc := null;
  end if;
  new.updated_at := now();
  return new;
end
$$;
revoke all on function public.adobe_sign_connection_rules() from public, anon, authenticated;

do $$
begin
  if not exists (select 1 from pg_trigger where tgname = 'adobe_sign_connection_rules_trg'
      and tgrelid = 'public.adobe_sign_connection'::regclass) then
    create trigger adobe_sign_connection_rules_trg
      before insert or update on public.adobe_sign_connection
      for each row execute function public.adobe_sign_connection_rules();
  end if;
end
$$;

-- OPEN ITEM C: 0106's append-only guard, copied from 0106 (lines 423-446) with ONE change: the FK's own ON DELETE SET NULL may
-- also clear `esign_request_id` (the owner's family delete removes the request by cascade). Nothing else may change.
create or replace function public.client_proposal_signature_guard()
returns trigger
language plpgsql
set search_path = ''
as $$
begin
  if tg_op = 'UPDATE' then
    -- The FK's own ON DELETE SET NULL (template_id, and from 0113 esign_request_id) arrives from inside the
    -- referential-action trigger; nothing else may change.
    if pg_trigger_depth() > 1
       and (to_jsonb(new) - 'template_id' - 'esign_request_id') = (to_jsonb(old) - 'template_id' - 'esign_request_id')
       and (new.template_id is null or new.template_id is not distinct from old.template_id)
       and (new.esign_request_id is null or new.esign_request_id is not distinct from old.esign_request_id) then
      return new;
    end if;
    raise exception 'loe_signature_append_only: a signature record is never changed'
      using errcode = '23514';
  end if;
  -- INSERT: the server clock, whatever the caller sent. (OPEN ITEM B: PART 7 proposes the Adobe exception.)
  new.signed_at := now();
  return new;
end
$$;
revoke all on function public.client_proposal_signature_guard() from public, anon, authenticated;

-- ── 4. client_proposal_esign_open (service role) ─────────────────────────────────────────────────────────────────────
-- Replaces a plain insert (D21): locks the proposal, checks it is still the same live version with a letter and that the
-- connection is this proposal's firm's and active, checks the sha256 of the text, and inserts the request as `creating`
-- with org_id = the PROPOSAL's. A second open request for the version fails on the partial unique index (23505).
create or replace function public.client_proposal_esign_open(
  p_proposal_id           uuid,
  p_org_id                uuid,
  p_family_id             uuid,
  p_version               integer,
  p_connection_id         uuid,
  p_signer_name           text,
  p_signer_title          text,
  p_entity_name           text,
  p_address               jsonb,
  p_template_version      integer,
  p_details               jsonb,
  p_signer_email          text,
  p_selection             jsonb,
  p_snapshot              jsonb,
  p_agreement_text        text,
  p_agreement_sha256      text,
  p_canonical_fingerprint text,
  p_sent_pdf_sha256       text
)
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_row  public.client_proposal%rowtype;
  v_conn record;
  v_id   uuid;
begin
  select * into v_row from public.client_proposal p where p.id = p_proposal_id for update;
  if not found
     or v_row.org_id is distinct from p_org_id
     or v_row.family_id is distinct from p_family_id
     or v_row.version is distinct from p_version
     or v_row.status is distinct from 'sent'
     -- 0106's expiry backstop (the app checks the firm's own calendar exactly).
     or (v_row.expires_on is not null and v_row.expires_on < (now() at time zone 'Etc/GMT+12')::date)
     or (jsonb_typeof(v_row.content -> 'loe') = 'object') is not true then
    raise exception 'esign_proposal_changed: the proposal is not the live version that was signed'
      using errcode = '23514';
  end if;

  select c.org_id, c.status into v_conn from public.adobe_sign_connection c where c.id = p_connection_id;
  if not found or v_conn.org_id is distinct from v_row.org_id or v_conn.status is distinct from 'active' then
    raise exception 'esign_proposal_changed: the Adobe connection is not this firm''s active one'
      using errcode = '23514';
  end if;

  if p_agreement_text is null or p_agreement_sha256 is null
     or pg_catalog.encode(pg_catalog.sha256(pg_catalog.convert_to(p_agreement_text, 'UTF8')), 'hex') is distinct from p_agreement_sha256 then
    raise exception 'esign_proposal_changed: the agreement text does not match its fingerprint'
      using errcode = '23514';
  end if;

  insert into public.client_proposal_esign_request (
    org_id, proposal_id, family_id, version, connection_id, provider, signer_name, signer_title, entity_name, address,
    template_version, canonical_fingerprint, agreement_text, agreement_sha256, sent_pdf_sha256, selection, snapshot, details,
    signer_email, status
  ) values (
    v_row.org_id, v_row.id, v_row.family_id, v_row.version, p_connection_id, 'adobe_sign', btrim(p_signer_name),
    nullif(btrim(coalesce(p_signer_title, '')), ''), nullif(btrim(coalesce(p_entity_name, '')), ''), p_address,
    p_template_version, p_canonical_fingerprint, p_agreement_text, p_agreement_sha256, p_sent_pdf_sha256, p_selection, p_snapshot,
    p_details, p_signer_email, 'creating'
  )
  returning id into v_id;
  return v_id;
end
$$;

revoke all on function public.client_proposal_esign_open(
  uuid, uuid, uuid, integer, uuid, text, text, text, jsonb, integer, jsonb, text, jsonb, jsonb, text, text, text, text)
  from public, anon, authenticated;
grant execute on function public.client_proposal_esign_open(
  uuid, uuid, uuid, integer, uuid, text, text, text, jsonb, integer, jsonb, text, jsonb, jsonb, text, text, text, text)
  to service_role;

-- ── 5. client_proposal_adobe_complete (service role) ─────────────────────────────────────────────────────────────────
-- D16: no p_org_id (the org is the locked request row's), no signer parameters (they are the request's frozen columns).
-- Returns (signature_id, completed_now). OPEN ITEM A: SECURITY INVOKER, for the same reason 0106 gives for sign_and_accept.
-- NO EXCEPTION BLOCK, NO SAVEPOINT (0106: the lifecycle looks for a signature row with this transaction's own xid).
create or replace function public.client_proposal_adobe_complete(
  p_request_id       uuid,
  p_agreement_id     text,
  p_signed_at        timestamptz,
  p_agreement_text   text,
  p_agreement_sha256 text
)
returns table (signature_id uuid, completed_now boolean)
language plpgsql
security invoker
set search_path = ''
as $$
declare
  v_req          public.client_proposal_esign_request%rowtype;
  v_row          public.client_proposal%rowtype;
  v_signature_id uuid := gen_random_uuid();
begin
  -- Lock order: the request first, then the proposal.
  select * into v_req from public.client_proposal_esign_request r where r.id = p_request_id for update;
  if not found then
    raise exception 'esign_request_closed: no such signing request' using errcode = '23514';
  end if;
  if v_req.status = 'completed' then
    if v_req.adobe_agreement_id is not distinct from p_agreement_id then
      return query select v_req.signature_id, false;
      return;
    end if;
    raise exception 'esign_request_closed: this request was completed by another agreement' using errcode = '23514';
  end if;

      using errcode = '23514';
  end if;

  -- signature is refused (one version is never signed twice). client_proposal_esign_open takes the same lock first.

     where r.proposal_id = v_row.id and r.status in ('creating', 'out_for_signature')

      using errcode = '23514';
  end if;

  -- update's fence) comes from v_row, never from the argument.
  if v_row.org_id is distinct from p_org_id then

      using errcode = '23514';
  end if;
  if v_row.status is distinct from 'sent' then
    raise exception 'loe_sign_not_sent: only a live proposal can be signed (this one is %)', v_row.status
      using errcode = '23514';
  end if;

  -- which the app checks exactly. Here it is expired only when that day is over
  -- everywhere on earth (the date in UTC-12).
  if v_row.expires_on is not null
     and v_row.expires_on < (now() at time zone 'Etc/GMT+12')::date then
    raise exception 'loe_sign_expired: this offer has expired'
      using errcode = '23514';
  end if;
  -- The same test the lifecycle trigger uses: the letter is a JSON object. A
  -- missing key and a JSON null both read as "no letter".
  if (jsonb_typeof(v_row.content -> 'loe') = 'object') is not true then
    raise exception 'loe_sign_no_letter: this proposal carries no engagement letter'
      using errcode = '23514';
  end if;

  end if;

  end if;

  -- (approved, or retired since) and is the very version the content names. A
  -- draft, another firm's template or a version that does not match is never
  -- linked: template_id is stored null and the signing still succeeds, because

    else
      v_template_id := null;
    end if;
  end if;

  insert into public.client_proposal_signature (
    id, org_id, proposal_id, family_id, version, signer_name, signer_title, entity_name,
    address, signer_email, ip, user_agent, agreement_text, agreement_sha256,
    template_id, template_version
  ) values (
    v_signature_id, v_row.org_id, v_row.id, v_row.family_id, v_row.version, btrim(p_signer_name),
    nullif(btrim(coalesce(p_signer_title, '')), ''), nullif(btrim(coalesce(p_entity_name, '')), ''),
    p_address, v_row.client_email, p_ip, left(p_user_agent, 400), p_agreement_text, p_agreement_sha256,
    v_template_id, v_template_ver
  )
  returning signed_at into v_signed_at;

  -- The marker the lifecycle trigger looks for, with this transaction's own
  -- signature row. Local to the transaction.
  perform set_config('lectual.loe_signing', v_row.id::text, true);

  update public.client_proposal p
     set status = 'accepted',
         accepted_selection = p_selection,
         accepted_snapshot = p_snapshot || jsonb_build_object(
           'schema', 2,
           'agreement', jsonb_build_object(
             'signature_id', v_signature_id,
             'signer_name', btrim(p_signer_name),
             'signed_at', v_signed_at,
             'sha256', p_agreement_sha256,
             'template_version', v_template_ver
           )
         )
   where p.id = v_row.id and p.org_id = v_row.org_id and p.status = 'sent';
  if not found then
    raise exception 'loe_sign_not_sent: only a live proposal can be signed'
      using errcode = '23514';
  end if;

  perform set_config('lectual.loe_signing', '', true);
  return v_signature_id;
end
$$;
