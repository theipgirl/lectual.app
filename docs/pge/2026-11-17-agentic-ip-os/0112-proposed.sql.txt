-- WHAT WAS CHECKED (PGE Sprint 4, Nov 20):
--   * Checked against lectual 0103 line 98 via the director: `constraint queue_card_event_kind check (kind in ('handoff','later',
--     'returned'))`. So `kind` has a CHECK (not an enum) and its real name is `queue_card_event_kind` (NOT `..._kind_check`).
--     The databases themselves were not reachable from the session that wrote this; run `\d public.queue_card_event` on
--     lectual-dev before applying.
--   * What this app shows: the hand-added type (src/lib/db/types.ts) has `kind: string`; the app writes and reads the kinds
--     'handoff', 'later', 'returned' (src/lib/queue/deck.ts `CardEventKind`, card-state.ts `KINDS`) and, with this file, 'draft_to'
--     and 'drafted'. AGENTS.md records (verified on lectual-prod 2026-11-12) that `actor_id` is forced to auth.uid(), a BEFORE
--     INSERT trigger sets created_at, signed-in users may insert only org_id, queue_item_id, kind, target_user_id, until, note,
--     and there is no UPDATE or DELETE.
--   * 0103 has TWO constraints that refuse the new kinds: `queue_card_event_kind` (line 98) and `queue_card_event_shape`
--     (lines ~100-104, only the three old shapes). This file replaces both under their own names. Before it, an insert of a new
--     kind answers 23514 naming one of them, which src/lib/queue/card-state.ts (`isKindNotSetUpError`) reads, BY CODE and
--     quote-delimited name (`"queue_card_event_kind"` or `"queue_card_event_shape"`), and only for the new kinds, as "not
--     switched on". The attorney's approve then falls back to today's behaviour for HER OWN mailbox only (never a third
--     person's), and the page says passing to the drafter isn't switched on yet.
--   * Parts 3 and 4 (who may write each kind; one claim per attempt, one failure per attempt, one drafted per item) are what
--     stop two concurrent "Make the draft" clicks and forged rows in the database; without them the app's fold (`trustedEvents`)
--     is the only guard.
--   * 0103's trigger `queue_card_event_rules()` (BEFORE INSERT OR UPDATE, `queue_card_event_rules_trg`) has branches for handoff,
--     later and returned only; any other kind falls through to the kind CHECK and gets `created_at := now()`. So the new kinds
--     pass the trigger untouched, and THIS FILE LEAVES THE TRIGGER ALONE (option chosen: no trigger change). Nothing below needs
--     a trigger rule: the shapes are the CHECK, who may write is the restrictive policy. (If a later change ever needs a branch
--     for the new kinds, it must `create or replace` the function with 0103's body copied exactly; that body is not in this
--     session, so do not write one from memory.) `actor_id` stays forced to auth.uid() and `created_at` stays the database's.
--   * 0103's `queue_card_event_not_self` (`target_user_id is null or target_user_id <> actor_id`) refuses a draft_to to oneself.
--     The app never writes one: the drafter approving her own item is plan `self`, which writes no draft_to at all, and a pass
--     targets the drafter only when the drafter is someone else (pinned by tests). `drafting`, `drafted`, `draft_failed` and
--     `pass_failed` have no target, so they pass it.
--   * 0103's insert policy `queue_card_event_insert_staff` (owner, admin, senior_admin, attorney, intake, paralegal, law_clerk,
--     clerk; `actor_id = auth.uid()` and `org_id = current_org_id()`) STAYS. The RESTRICTIVE policy below sits on top of it:
--     a new-kind row must pass both.
--   * The attempt is a real column (step 0 below), not the note: 0103's insert grant is the column list
--     (org_id, queue_item_id, kind, target_user_id, until, note), and `note` is free text any staff member writes, so a unique
--     index over it would be fragile. The app sends `attempt` only for drafting / drafted / draft_failed and never parses `note`.
--
-- Why (decision 3b): when the attorney approves a client email and the firm's "Drafts go to" is someone else, the card passes
-- to that person (`draft_to`, target = the drafter, note = 'approved-text sha256 <64 hex>') and they make the draft in their OWN
-- Outlook on their own click: `drafting` (the claim) first, then `drafted` (confirmed) or `draft_failed` (open again). Nobody's
-- mailbox is written without their own click. A failed pass at approval leaves a `pass_failed` marker for the approver's retry.
-- =====================================================================================================================

begin;

-- APPLYING: the `drop constraint` statements below may hang the Supabase MCP when sent as one migration; apply this file in
-- execute_sql chunks instead (see 0093).
--
-- 0. The drafter's attempt number, a real column (null for every other kind; required, >= 1, for the drafting kinds: see the
--    shape CHECK), and the insert grant extended to it. The app reads it with a fallback to 0103's columns before this file.
alter table public.queue_card_event add column if not exists attempt smallint;
grant insert (attempt) on public.queue_card_event to authenticated;

-- 1. The five new kinds. Drop and recreate 0103's kind CHECK under its real name, `queue_card_event_kind` (0103 line 98).
--      draft_to      the approver passes an approved email to the firm's drafter (target = the drafter)
--      drafting      the drafter CLAIMS attempt <n> (the `attempt` column) before making the Outlook draft
--      drafted       the draft is CONFIRMED made, written only after Outlook said so (same `attempt`)
--      draft_failed  attempt <n> made no draft; the item is open again for the same drafter (same `attempt`)
--      pass_failed   the approve path couldn't write the draft_to: the marker that lets the approver retry the pass
alter table public.queue_card_event drop constraint if exists queue_card_event_kind;
alter table public.queue_card_event
  add constraint queue_card_event_kind
  check (kind in ('handoff', 'later', 'returned', 'draft_to', 'drafting', 'drafted', 'draft_failed', 'pass_failed'));

-- 2. Shapes. 0103 adds `queue_card_event_shape`, which admits ONLY the three old shapes, so the new kinds would still be refused
--    after step 1. Drop it and re-add it under the SAME name: 0103's three clauses verbatim (0103 lines 100-104), each now also
--    requiring `attempt is null`, plus the new shapes.
alter table public.queue_card_event drop constraint if exists queue_card_event_shape;
alter table public.queue_card_event
  add constraint queue_card_event_shape
  check (
    ((kind = 'handoff'  and target_user_id is not null and until is null)
    or (kind = 'later'    and target_user_id is null     and until is not null)
    or (kind = 'returned' and target_user_id is null     and until is null))
      and attempt is null
    or (kind = 'draft_to'    and target_user_id is not null and until is null and attempt is null)
    or (kind = 'pass_failed' and target_user_id is null     and until is null and attempt is null)
    or (kind in ('drafting', 'drafted', 'draft_failed') and target_user_id is null and until is null
        and attempt is not null and attempt >= 1)
  );

-- 3. Who may insert them. 0103's insert policy (staff of the row's org) stays; this adds, for the new kinds only:
--    * draft_to:     the TARGET is the firm's configured drafter (client_proposal_setting.pricing->'approval_routing'->>'drafter',
--                    the same setting the app reads), and the WRITER holds owner / admin / senior_admin / attorney or is named
--                    as a value in that firm's approval_routing (a routed approver). Same org only.
--    * pass_failed:  the same writer rule as draft_to (no target).
--    * drafting:     only the CURRENT draft_to target (the newest draft_to on the item, by created_at then id), for themselves
--                    (actor_id is auth.uid() by the trigger), and only for the NEXT attempt: attempt = (the number of their
--                    draft_failed rows on the item since that draft_to) + 1, and never once a drafted exists.
--    * drafted / draft_failed: only the current target, and only for their own newest `drafting` attempt that has no outcome yet.
--    RESTRICTIVE, so it narrows 0103's permissive policy and widens nothing.
--
--    The helper is SECURITY DEFINER (it reads crm_org_member, client_proposal_setting and other rows of queue_card_event that
--    RLS would hide), so it lives in `private`, a schema PostgREST does NOT expose (check `pgrst.db_schemas` / the API
--    settings: only `public` and `graphql_public` should be listed). It is not callable as an RPC. Its first statement refuses
--    any org but the caller's own active one, so even a direct call can't answer about another firm. `authenticated` gets
--    USAGE on the schema and EXECUTE on this one function only because the policy expression runs as the caller.
create schema if not exists private;
revoke all on schema private from public, anon;
grant usage on schema private to authenticated;

create or replace function private.queue_card_event_draft_ok(p_org uuid, p_item text, p_kind text, p_target uuid, p_attempt smallint)
returns boolean
language plpgsql
stable
security definer
set search_path = ''
as $$
declare
  v_routing jsonb;
  v_newest_target uuid;
  v_newest_at timestamptz;
  v_failed int;
  v_open_claim smallint;
begin
  if p_org is distinct from public.current_org_id() then
    return false;
  end if;

  if p_kind in ('draft_to', 'pass_failed') then
    select s.pricing -> 'approval_routing' into v_routing
    from public.client_proposal_setting s
    where s.org_id = p_org;
    if v_routing is null then
      return false;
    end if;
    if p_kind = 'draft_to' and (p_target is null or p_target::text is distinct from (v_routing ->> 'drafter')) then
      return false;
    end if;
    -- The writer: an approver role in this firm, or a person the firm routes approvals to.
    return exists (
      select 1 from public.crm_org_member m
      where m.org_id = p_org and m.user_id = auth.uid()
        and (
          m.role::text in ('owner', 'admin', 'senior_admin', 'attorney')
          or exists (select 1 from jsonb_each_text(v_routing) r where r.value = auth.uid()::text)
        )
    );
  end if;

  if p_kind in ('drafting', 'drafted', 'draft_failed') then
    select e.target_user_id, e.created_at into v_newest_target, v_newest_at
    from public.queue_card_event e
    where e.org_id = p_org and e.queue_item_id = p_item and e.kind = 'draft_to'
    order by e.created_at desc, e.id desc
    limit 1;
    if v_newest_target is null or v_newest_target <> auth.uid() then
      return false;
    end if;
    if exists (select 1 from public.queue_card_event e where e.org_id = p_org and e.queue_item_id = p_item
               and e.kind = 'drafted' and e.created_at >= v_newest_at) then
      return false;                                                  -- confirmed already: nothing more
    end if;
    if p_kind = 'drafting' then
      select count(*) into v_failed from public.queue_card_event e
      where e.org_id = p_org and e.queue_item_id = p_item and e.kind = 'draft_failed'
        and e.actor_id = auth.uid() and e.created_at >= v_newest_at;
      return p_attempt = v_failed + 1;                               -- only the next attempt can be claimed
    end if;
    -- drafted / draft_failed: the caller's own newest claim, still without an outcome.
    select c.attempt into v_open_claim from public.queue_card_event c
    where c.org_id = p_org and c.queue_item_id = p_item and c.kind = 'drafting' and c.actor_id = auth.uid()
      and c.created_at >= v_newest_at
      and not exists (select 1 from public.queue_card_event f where f.org_id = p_org and f.queue_item_id = p_item
                      and f.kind = 'draft_failed' and f.actor_id = c.actor_id and f.attempt = c.attempt)
    order by c.created_at desc, c.id desc
    limit 1;
    return v_open_claim is not null and p_attempt = v_open_claim;
  end if;
  return true;
end
$$;
revoke all on function private.queue_card_event_draft_ok(uuid, text, text, uuid, smallint) from public, anon;
grant execute on function private.queue_card_event_draft_ok(uuid, text, text, uuid, smallint) to authenticated;

drop policy if exists queue_card_event_draft_kinds on public.queue_card_event;
create policy queue_card_event_draft_kinds on public.queue_card_event
  as restrictive
  for insert
  to authenticated
  with check (
    kind not in ('draft_to', 'drafting', 'drafted', 'draft_failed', 'pass_failed')
    or (org_id = public.current_org_id() and private.queue_card_event_draft_ok(org_id, queue_item_id, kind, target_user_id, attempt))
  );

-- 4. Two concurrent clicks make at most ONE Outlook draft:
--    * A click CLAIMS attempt n with a `drafting` row BEFORE it calls Outlook. One `drafting` per (org, item, actor, attempt):
--      attempts are numbered PER DRAFTER (the helper counts the caller's own draft_failed rows), so a new drafter's attempt 1
--      never collides with a departed drafter's attempt 1. Both
--      clicks compute the same n from the rows they read, so the second insert answers 23505 and the app refuses it ("already
--      being drafted"). The policy only admits the NEXT attempt, so nobody can pre-claim attempt 2 while attempt 1 is open.
--    * After a `draft_failed` for attempt n, the next claim is n + 1, a new key: a retry after a failure is allowed.
--    * One `draft_failed` per attempt and one `drafted` per item (confirmation only after Outlook said so).
create unique index if not exists queue_card_event_one_claim_per_attempt
  on public.queue_card_event (org_id, queue_item_id, actor_id, attempt)
  where kind = 'drafting';
create unique index if not exists queue_card_event_one_failure_per_attempt
  on public.queue_card_event (org_id, queue_item_id, actor_id, attempt)
  where kind = 'draft_failed';
create unique index if not exists queue_card_event_one_drafted
  on public.queue_card_event (org_id, queue_item_id)
  where kind = 'drafted';
create unique index if not exists queue_card_event_one_pass_failed
  on public.queue_card_event (org_id, queue_item_id)
  where kind = 'pass_failed';

commit;
