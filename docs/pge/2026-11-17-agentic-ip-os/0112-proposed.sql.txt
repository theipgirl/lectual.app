--   * IF 0103 HAS NO CHECK AND NO ENUM ON `kind`: the two kinds already insert, this file is not needed for the kinds, and the
--     app works without it. Parts 2 to 4 (target = the firm's configured drafter, writer an approver role or routed, drafted

--     "Make the draft" clicks are not stopped by the database, and without parts 2 and 3 the app's fold (`trustedEvents`) is
--     the only thing standing between a forged row and the Ready list.
--   * The app recognises "before 0112" only by code AND message: 23514 naming `queue_card_event_kind_check`, or 22P02 "invalid
--     input value for enum" naming the new kind (card-state.ts `isKindNotSetUpError`).
--
-- Why (decision 3b): when the attorney approves a client email and the firm's "Drafts go to" is someone else, the card passes
-- to that person (`draft_to`, target = the drafter, note = 'approved-text sha256 <64 hex>') and they make the draft in their OWN
-- Outlook on their own click (`drafted`, no target). Nobody's mailbox is written without their own click.

begin;

-- 1. The two kinds. Replace the constraint name with 0103's real one (`\d public.queue_card_event` on lectual-dev).
--    If 0103 used an enum instead: `alter type public.<enum> add value if not exists 'draft_to'; ... 'drafted';` (outside a
--    transaction on older Postgres) and skip this step.
alter table public.queue_card_event drop constraint if exists queue_card_event_kind_check;
alter table public.queue_card_event
  add constraint queue_card_event_kind_check
  check (kind in ('handoff', 'later', 'returned', 'draft_to', 'drafted'));

-- 2. Shape rules for the new kinds (the old kinds are untouched).
alter table public.queue_card_event drop constraint if exists queue_card_event_draft_shape_check;
alter table public.queue_card_event
  add constraint queue_card_event_draft_shape_check
  check (
    (kind <> 'draft_to' or target_user_id is not null)   -- a draft_to names its drafter
    and (kind <> 'drafted' or target_user_id is null)    -- a drafted names nobody: it is the actor's own
  );

-- 3. Who may insert them. 0103's insert policy (staff of the row's org) stays; this adds, for the new kinds only:
--    * draft_to: the TARGET is the firm's configured drafter (client_proposal_setting.pricing->'approval_routing'->>'drafter',
--                the same setting the app reads), and the WRITER holds owner / admin / senior_admin / attorney or is named as a
--                value in that firm's approval_routing (a routed approver). Same org only.
--    * drafted:  only for oneself (actor_id is auth.uid() by the trigger), and only when the newest draft_to on that item
--                (newest by created_at, then id) names the caller.
--    RESTRICTIVE, so it narrows 0103's permissive policy and widens nothing.
--
--    The helper is SECURITY DEFINER (it reads crm_org_member, client_proposal_setting and other rows of queue_card_event that
--    RLS would hide), so it lives in `private`, a schema PostgREST does NOT expose (check `pgrst.db_schemas` / the API
--    settings: only `public` and `graphql_public` should be listed). It is not callable as an RPC. Its first statement refuses
--    any org but the caller's own active one, so even a direct call can't answer about another firm. `authenticated` gets
--    USAGE on the schema and EXECUTE on this one function only because the policy expression runs as the caller.
create schema if not exists private;
revoke all on schema private from public, anon;
grant usage on schema private to authenticated;

create or replace function private.queue_card_event_draft_ok(p_org uuid, p_item text, p_kind text, p_target uuid)
returns boolean
language plpgsql
stable
security definer
set search_path = ''
as $$
declare
  v_routing jsonb;
  v_newest_target uuid;
begin
  if p_org is distinct from public.current_org_id() then
    return false;
  end if;

  if p_kind = 'draft_to' then
    select s.pricing -> 'approval_routing' into v_routing
    from public.client_proposal_setting s
    where s.org_id = p_org;
    if v_routing is null or p_target is null or p_target::text is distinct from (v_routing ->> 'drafter') then
      return false;
    end if;
    -- The writer: an approver role in this firm, or a person the firm routes approvals to.
    return exists (
      select 1 from public.crm_org_member m
      where m.org_id = p_org and m.user_id = auth.uid()
        and (
          m.role::text in ('owner', 'admin', 'senior_admin', 'attorney')
          or exists (select 1 from jsonb_each_text(v_routing) r where r.value = auth.uid()::text)
        )
    );
  elsif p_kind = 'drafted' then
    select e.target_user_id into v_newest_target
    from public.queue_card_event e
    where e.org_id = p_org and e.queue_item_id = p_item and e.kind = 'draft_to'
    order by e.created_at desc, e.id desc
    limit 1;
    return v_newest_target is not null and v_newest_target = auth.uid();
  end if;
  return true;
end
$$;
revoke all on function private.queue_card_event_draft_ok(uuid, text, text, uuid) from public, anon;
grant execute on function private.queue_card_event_draft_ok(uuid, text, text, uuid) to authenticated;

drop policy if exists queue_card_event_draft_kinds on public.queue_card_event;
create policy queue_card_event_draft_kinds on public.queue_card_event
  as restrictive
  for insert
  to authenticated
  with check (
    kind not in ('draft_to', 'drafted')
    or (org_id = public.current_org_id() and private.queue_card_event_draft_ok(org_id, queue_item_id, kind, target_user_id))
  );

-- 4. One `drafted` per item: the app writes it BEFORE it makes the Outlook draft (a claim), so two clicks make at most one
--    draft. The second insert answers 23505, which the app reads as "already drafted".
create unique index if not exists queue_card_event_one_drafted
  on public.queue_card_event (org_id, queue_item_id)
  where kind = 'drafted';

commit;
