-- ── THE GAP ──────────────────────────────────────────────────────────────────────────────────────────────────────────
-- 0002_rls.sql, verbatim:
--   create policy member_insert_admin on crm_org_member
--     for insert with check (
--       org_id = public.current_org_id()
--       and public.current_org_role() in ('owner','admin','senior_admin')
--     );
--   create policy member_update_admin on crm_org_member
--     for update using (
--       org_id = public.current_org_id()
--       and public.current_org_role() in ('owner','admin','senior_admin')
--     ) with check (
--       org_id = public.current_org_id()
--       and public.current_org_role() in ('owner','admin','senior_admin')
--     );
--   create policy member_delete_admin on crm_org_member
--     for delete using (
--       org_id = public.current_org_id()
--       and public.current_org_role() in ('owner','admin','senior_admin')
--     );
-- (plus member_select_own_org, and 0005's member_select_auth_admin for the token hook.) All three write policies trust the JWT
-- role CLAIM and nothing else, and nothing compares the old row with the new one. So, through PostgREST:
--   1. A Lectual platform admin (0050) inside a firm they do not belong to carries `org_role = owner` (0050's token hook,
--      "org_role is 'owner' so the firm's role-gated RLS write policies behave as they would for the firm's own principal"). They
--      can INSERT (firm, self, 'owner'), and after a token refresh the hook finds that row and they are a REAL owner of the firm,
--      with no further platform_admin_audit entry. Self-inserting as admin or senior_admin gives admin-tier rights the same way.
--   2. An admin or senior_admin can UPDATE their own row to owner (or admin -> senior_admin -> anything), and can change, demote or
--      DELETE an existing owner's row, and can insert a new owner row for anyone (an accomplice, a second account).
--   3. The UPDATE policy also lets an admin rewrite a row's user_id (move a paralegal's row onto themselves) or created_at.
--   4. The "never leave the firm ownerless" rule exists only in the app (src/lib/settings/admin.ts, members.ts).
--   5. The claim outlives the row: an admin who was demoted or removed keeps admin-tier membership writes until their token
--      expires (up to an hour; 0031 revokes sessions on removal, but a token already issued still works until it expires).
-- The app refuses most of these in its own checks (no self-change, rank ceilings, last-owner guard), but RLS is the boundary and
-- anyone holding a session token can call PostgREST directly.
--
-- ── THE RULE (Taylor, Nov 23) ────────────────────────────────────────────────────────────────────────────────────────
-- For a SIGNED-IN caller (any role that is not trusted, below), on public.crm_org_member:
--   a. Nobody inserts their OWN row, whatever the role (member_owner_guard_self_insert). Recommended over "only not as owner": a
--      real member already has a row (unique (org_id, user_id)), so the only callers who can ever self-insert are non-members
--      holding an admin-tier claim, i.e. platform admins, and joining a firm as admin is an escalation too. No real flow
--      self-inserts through the API: a new firm's first owner is made by provision_org / provision_org_for_owner (below).
--   b. Nobody changes their OWN role, to anything (member_owner_guard_self_role). Wider than "never to owner" on purpose: an admin
--      promoting themselves to senior_admin, or to attorney (0110's close override reads the attorney membership row), is an
--      escalation too, and the app already refuses every self-change ("You can't change your own role.").
--   c. EVERY membership write needs a REAL admin-tier row for the caller in the active firm (owner, admin or senior_admin, read from
--      crm_org_member, not the claim), or the caller is Lectual staff (public.is_platform_admin(), 0050)
--      (member_owner_guard_not_admin). This closes gap 5: a demoted or removed admin's still-valid token no longer writes membership.
--   d. Inserting an owner row, changing a row to or from owner, or deleting an owner row needs a REAL owner: a crm_org_member row
--      (org_id = the active firm, user_id = auth.uid(), role = 'owner'). That is a table lookup, never current_org_role(), so a
--      platform admin's `owner` claim does not count and a demoted owner's stale token does not count (member_owner_guard_not_owner).
--   e. Only `role` may change on UPDATE: every other column, today's and any added later, is compared as
--      `(to_jsonb(new) - 'role') is distinct from (to_jsonb(old) - 'role')` (0115's form) (member_owner_guard_immutable).
--   f. The firm keeps at least one owner after a signed-in update or delete (member_owner_guard_last_owner). Checked AFTER the
--      statement's rows are written (an AFTER ROW trigger sees the whole statement), under a per-firm advisory lock, so two concurrent
--      transactions each demoting one of two owners cannot both pass, and neither can one statement that removes both.
--   g. A row whose org_id is not the caller's active firm is refused with one fixed sentence before anything else is read
--      (member_owner_guard_wrong_org), so the guard is no oracle about another firm. RLS refuses those writes anyway.
--   h. No session (auth.uid() null) through an untrusted role: member_owner_guard_no_session.
-- All refusals are SQLSTATE 42501 with those stable prefixes. lectual.app's friendlySettingsError
-- (src/app/dashboard/settings/team/errors.ts) already maps any 42501 to "You don't have permission to do that.", so no database
-- text reaches a person and no app change is needed for the refusals themselves.
--
-- DELETE: 0002 has member_delete_admin (above). Deleting an owner row is "touching" one, so it needs a real owner (rule d), and a
-- signed-in delete of the last owner is refused (rule f). Deleting a non-owner row needs a real admin-tier row (rule c). Leaving a
-- firm by deleting your own non-owner row is allowed only for the admin tier (rule c; the app refuses self-removal anyway).
--
-- ── WHO IS TRUSTED (passes untouched) ────────────────────────────────────────────────────────────────────────────────
--   * Rows whose parent is gone. Deleting a crm_org, or an auth.users row (Supabase GoTrue, which deletes as supabase_auth_admin),
--     cascades into crm_org_member. The BEFORE trigger sees that delete at pg_trigger_depth() > 1 (0085's and 0107's form) and lets
--     it through. The AFTER ROW trigger does NOT: the cascade's AFTER row events are run at the end of the outer statement, at
--     depth 1, with current_user = the outer role (supabase_auth_admin through GoTrue), which is not trusted. So the AFTER guard's
--     first test, for a DELETE, is private.crm_org_member_parent_gone(old.org_id, old.user_id): true when the crm_org row OR the
--     auth.users row no longer exists, and then it passes. Consequence (Taylor's choice): deleting the auth user of a firm's only
--     owner succeeds and leaves the firm with no owner; a signed-in "delete the only owner" still gets last_owner (both parents
--     exist then). No trigger on any table reachable by an API role writes crm_org_member, so depth > 1 is never a way in for a
--     signed-in caller; a future trigger that does must re-check.
--   * current_user = 'service_role' (scripts, the server's admin client, tests' seed helpers).
--   * current_user is the table's OWNER or a member of it (pg_has_role(current_user, relowner, 'MEMBER'); superusers are members of
--     every role). This is what a SECURITY DEFINER function owned by postgres runs as, so provision_org() and
--     provision_org_for_owner() keep creating a new firm's first owner. On Supabase the API roles are NOT members of postgres
--     (postgres is a member of authenticated, not the reverse); PRE-APPLY CHECK 1 confirms it, and that both provisioning
--     functions are owned by the table owner (or a member of it).
-- For that reason the two GUARD functions are SECURITY INVOKER: current_user must be the caller's role. The lookups that have to see
-- past RLS are two SECURITY DEFINER helpers in schema `private`, which PostgREST does not expose (0112's pattern; check
-- `pgrst.db_schemas` does not list it), so no member can call them over rpc or take the owner lock directly. Owned by the table
-- owner, they bypass RLS (the table is not FORCE ROW LEVEL SECURITY), so a guard's read never recurses through a crm_org_member
-- policy. private.crm_org_member_caller_facts(p_lock) answers only for auth.uid() in current_org_id() (the caller's real role, whether
-- they are Lectual staff, the firm's owner count). private.crm_org_member_parent_gone(org, user) answers only whether two rows exist.
-- `authenticated` gets USAGE on `private` and EXECUTE on the facts helper because the invoker trigger runs as it; the parent_gone
-- helper is also granted to service_role and supabase_auth_admin, the roles a cascade's AFTER event runs as.
--
-- WHY A TRIGGER, NOT ONLY TIGHTER POLICIES
--   A policy cannot compare OLD with NEW (self role change, immutables, "touches an owner row"), a policy reading crm_org_member
--   from a crm_org_member policy invites recursion, policies give no stable SQLSTATE prefix, and the last-owner rule needs the whole
--   statement's result. The 0002 policies are left exactly as they are: they still decide WHO may write membership at all (the
--   admin-tier claim) and the triggers only narrow it.
--
-- GRANT CHANGE (flagged): this file also revokes TRUNCATE, REFERENCES and TRIGGER (and MAINTAIN on PostgreSQL 17) on
-- crm_org_member from anon and authenticated, as every later tenant table does (0098, 0099, 0103, 0115). TRUNCATE bypasses row
-- triggers and RLS, so without the revoke an API role could still empty the table outside these guards if PostgREST ever exposed it.
-- Nothing in either app needs those privileges. SELECT / INSERT / UPDATE / DELETE are untouched.

-- ── 1. Schema `private` (not exposed by PostgREST; 0112's form) ────────────────────────────────────────────────────
create schema if not exists private;
revoke all on schema private from public;
revoke all on schema private from anon;
grant usage on schema private to authenticated;
grant usage on schema private to service_role;
do $$
begin
  if exists (select 1 from pg_catalog.pg_roles where rolname = 'supabase_auth_admin') then
    execute 'grant usage on schema private to supabase_auth_admin';
  end if;
end
$$;


-- ── 2. The lookups that must see past RLS ──────────────────────────────────────────────────────────────────────────
-- Answers for the CALLER in their ACTIVE firm only: their real role there (null = no membership row), whether they are Lectual staff,
-- and the firm's owner count. With p_lock it takes the firm's owner lock first and reads in later statements, so under READ COMMITTED
-- (PostgREST's level) each read sees everything committed before the lock was granted. The lock is held to the end of the
-- transaction. Volatile on purpose: each statement takes a fresh snapshot.
create or replace function private.crm_org_member_caller_facts(
  p_lock boolean,
  out caller_role text,
  out caller_is_platform_admin boolean,
  out owner_count integer
)
language plpgsql
volatile
security definer
set search_path = ''
as $$
declare
  v_org uuid := public.current_org_id();
  v_uid uuid := auth.uid();
begin
  caller_role := null;
  caller_is_platform_admin := false;
  owner_count := 0;
  if v_org is null or v_uid is null then
    return;
  end if;

  if p_lock then
    perform pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended('crm_org_member_owner:' || v_org::text, 0));
  end if;

  select m.role::text into caller_role
    from public.crm_org_member m
   where m.org_id = v_org and m.user_id = v_uid;

  caller_is_platform_admin := coalesce(public.is_platform_admin(), false);

  select count(*)::integer into owner_count
    from public.crm_org_member m
   where m.org_id = v_org and m.role = 'owner';
end;
$$;

revoke all on function private.crm_org_member_caller_facts(boolean) from public;
revoke all on function private.crm_org_member_caller_facts(boolean) from anon;
grant execute on function private.crm_org_member_caller_facts(boolean) to authenticated;

-- True when the membership row's firm OR its auth user no longer exists: the row is being removed by a cascade.
create or replace function private.crm_org_member_parent_gone(p_org uuid, p_user uuid)
returns boolean
language sql
stable
security definer
set search_path = ''
as $$
  select not exists (select 1 from public.crm_org o where o.id = p_org)
      or not exists (select 1 from auth.users u where u.id = p_user);
$$;

revoke all on function private.crm_org_member_parent_gone(uuid, uuid) from public;
revoke all on function private.crm_org_member_parent_gone(uuid, uuid) from anon;
grant execute on function private.crm_org_member_parent_gone(uuid, uuid) to authenticated;
grant execute on function private.crm_org_member_parent_gone(uuid, uuid) to service_role;
do $$
begin
  if exists (select 1 from pg_catalog.pg_roles where rolname = 'supabase_auth_admin') then
    execute 'grant execute on function private.crm_org_member_parent_gone(uuid, uuid) to supabase_auth_admin';
  end if;
end
$$;


-- ── 3. BEFORE guard: who may write which row ───────────────────────────────────────────────────────────────────────
-- SECURITY INVOKER (no security definer): current_user must be the caller's own role for the trust test.
create or replace function public.crm_org_member_owner_guard()
returns trigger
language plpgsql
set search_path = ''
as $$
declare
  v_uid uuid;
  v_org uuid;
  v_touches_owner boolean := false;
  v_facts record;
begin
  -- 3a. Referential actions (a crm_org or auth.users delete cascading here) pass, as in 0085 and 0107.
  if pg_catalog.pg_trigger_depth() > 1 then
    if tg_op = 'DELETE' then return old; end if;
    return new;
  end if;

  -- 3b. Trusted writers: the service role, and the table's owner (SECURITY DEFINER functions such as provision_org, migrations).
  if current_user = 'service_role'
     or exists (
       select 1 from pg_catalog.pg_class c
        where c.oid = 'public.crm_org_member'::regclass
          and pg_catalog.pg_has_role(current_user, c.relowner, 'MEMBER')
     )
  then
    if tg_op = 'DELETE' then return old; end if;
    return new;
  end if;

  -- 3c. Everyone else must be signed in.
  v_uid := auth.uid();
  if v_uid is null then
    raise exception 'member_owner_guard_no_session: membership changes need a signed-in caller'
      using errcode = '42501';
  end if;

  -- 3d. Only the active firm's rows, refused before anything is read (no oracle about another firm).
  v_org := public.current_org_id();
  if (tg_op in ('INSERT', 'UPDATE') and new.org_id is distinct from v_org)
     or (tg_op in ('UPDATE', 'DELETE') and old.org_id is distinct from v_org)
  then
    raise exception 'member_owner_guard_wrong_org: membership can only change in your active firm'
      using errcode = '42501';
  end if;

  -- 3e. Only the role changes on an update (every other column, today's and any added later).
  if tg_op = 'UPDATE' and (pg_catalog.to_jsonb(new) - 'role') is distinct from (pg_catalog.to_jsonb(old) - 'role') then
    raise exception 'member_owner_guard_immutable: only a member''s role can change'
      using errcode = '42501';
  end if;

  -- 3f. Never your own row: no self-insert (any role), no self role change (any role).
  if tg_op = 'INSERT' and new.user_id = v_uid then
    raise exception 'member_owner_guard_self_insert: you cannot add yourself to a firm'
      using errcode = '42501';
  end if;
  if tg_op = 'UPDATE' and old.user_id = v_uid and new.role is distinct from old.role then
    raise exception 'member_owner_guard_self_role: you cannot change your own role'
      using errcode = '42501';
  end if;

  -- 3g. Does this write create, change or remove an owner row?
  if tg_op = 'INSERT' then
    v_touches_owner := new.role = 'owner';
  elsif tg_op = 'UPDATE' then
    v_touches_owner := new.role is distinct from old.role and (old.role = 'owner' or new.role = 'owner');
  else
    v_touches_owner := old.role = 'owner';
  end if;

  -- 3h. The caller's REAL standing in this firm (the table, not the claim). The owner lock is taken only when an owner row is touched.
  select f.caller_role, f.caller_is_platform_admin, f.owner_count into v_facts
    from private.crm_org_member_caller_facts(v_touches_owner) f;

  if not (coalesce(v_facts.caller_role, '') in ('owner', 'admin', 'senior_admin') or coalesce(v_facts.caller_is_platform_admin, false)) then
    raise exception 'member_owner_guard_not_admin: only a current owner or admin of this firm can change its members'
      using errcode = '42501';
  end if;

  if v_touches_owner and coalesce(v_facts.caller_role, '') <> 'owner' then
    raise exception 'member_owner_guard_not_owner: only an owner of this firm can add, change or remove an owner'
      using errcode = '42501';
  end if;

  if tg_op = 'DELETE' then return old; end if;
  return new;
end;
$$;


-- ── 4. AFTER guard: the firm keeps at least one owner ──────────────────────────────────────────────────────────────
-- AFTER ROW triggers fire once the statement has written every row, so the count includes the whole statement (two owners removed in
-- one statement) and, after the lock, every transaction committed before it (two owners demoted in two transactions).
create or replace function public.crm_org_member_last_owner_guard()
returns trigger
language plpgsql
set search_path = ''
as $$
declare
  v_facts record;
begin
  -- 4a. A cascade from a deleted firm or auth user: its AFTER events run at depth 1 as the outer role, so test the parents.
  if tg_op = 'DELETE' and private.crm_org_member_parent_gone(old.org_id, old.user_id) then
    return null;
  end if;

  if pg_catalog.pg_trigger_depth() > 1 then
    return null;
  end if;

  if current_user = 'service_role'
     or exists (
       select 1 from pg_catalog.pg_class c
        where c.oid = 'public.crm_org_member'::regclass
          and pg_catalog.pg_has_role(current_user, c.relowner, 'MEMBER')
     )
  then
    return null;
  end if;

  -- Only a row that WAS an owner and no longer is one can leave the firm without an owner.
  if old.role is distinct from 'owner' then
    return null;
  end if;
  if tg_op = 'UPDATE' and new.role = 'owner' then
    return null;
  end if;

  if old.org_id is distinct from public.current_org_id() then
    raise exception 'member_owner_guard_wrong_org: membership can only change in your active firm'
      using errcode = '42501';
  end if;

  select f.owner_count into v_facts from private.crm_org_member_caller_facts(true) f;
  if coalesce(v_facts.owner_count, 0) < 1 then
    raise exception 'member_owner_guard_last_owner: a firm must keep at least one owner'
      using errcode = '42501';
  end if;

  return null;
end;
$$;

revoke execute on function public.crm_org_member_owner_guard() from public;
revoke execute on function public.crm_org_member_owner_guard() from anon;
revoke execute on function public.crm_org_member_owner_guard() from authenticated;
revoke execute on function public.crm_org_member_last_owner_guard() from public;
revoke execute on function public.crm_org_member_last_owner_guard() from anon;
revoke execute on function public.crm_org_member_last_owner_guard() from authenticated;


-- ── 5. Triggers ────────────────────────────────────────────────────────────────────────────────────────────────────
do $$
begin
  if not exists (
    select 1 from pg_trigger
     where tgrelid = 'public.crm_org_member'::regclass and tgname = 'crm_org_member_owner_guard_trg'
  ) then
    create trigger crm_org_member_owner_guard_trg
      before insert or update or delete on public.crm_org_member
      for each row execute function public.crm_org_member_owner_guard();
  end if;

  if not exists (
    select 1 from pg_trigger
     where tgrelid = 'public.crm_org_member'::regclass and tgname = 'crm_org_member_last_owner_guard_trg'
  ) then
    create trigger crm_org_member_last_owner_guard_trg
      after update or delete on public.crm_org_member
      for each row execute function public.crm_org_member_last_owner_guard();
  end if;
end
$$;

begin
  if pg_catalog.current_setting('server_version_num')::int >= 170000 then

  end if;
end
$$;
