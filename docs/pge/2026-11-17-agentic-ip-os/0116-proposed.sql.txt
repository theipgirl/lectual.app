-- ── THE GAP ──────────────────────────────────────────────────────────────────────────────────────────────────────────
-- 0002_rls.sql, verbatim:
--   create policy member_insert_admin on crm_org_member
--     for insert with check (
--       org_id = public.current_org_id()
--       and public.current_org_role() in ('owner','admin','senior_admin')
--     );
--   create policy member_update_admin on crm_org_member
--     for update using (
--       org_id = public.current_org_id()
--       and public.current_org_role() in ('owner','admin','senior_admin')
--     ) with check (
--       org_id = public.current_org_id()
--       and public.current_org_role() in ('owner','admin','senior_admin')
--     );
--   create policy member_delete_admin on crm_org_member
--     for delete using (
--       org_id = public.current_org_id()
--       and public.current_org_role() in ('owner','admin','senior_admin')
--     );
-- (plus member_select_own_org, and 0005's member_select_auth_admin for the token hook.) All three write policies trust the JWT
-- role CLAIM and nothing else, and nothing compares the old row with the new one. So, through PostgREST:
--   1. A Lectual platform admin (0050) inside a firm they do not belong to carries `org_role = owner` (0050's token hook,
--      "org_role is 'owner' so the firm's role-gated RLS write policies behave as they would for the firm's own principal"). They
--      can INSERT (firm, self, 'owner'), and after a token refresh the hook finds that row and they are a REAL owner of the firm,
--      with no further platform_admin_audit entry. Self-inserting as admin or senior_admin gives admin-tier rights the same way.
--   2. An admin or senior_admin can UPDATE their own row to owner (or admin -> senior_admin -> anything), and can change, demote or
--      DELETE an existing owner's row, and can insert a new owner row for anyone (an accomplice, a second account).
--   3. The UPDATE policy also lets an admin rewrite a row's user_id (move a paralegal's row onto themselves) or created_at.
--   4. The "never leave the firm ownerless" rule exists only in the app (src/lib/settings/admin.ts, members.ts).
--   5. The claim outlives the row: an admin who was demoted or removed keeps admin-tier membership writes until their token
--      expires (up to an hour; 0031 revokes sessions on removal, but a token already issued still works until it expires).
-- The app refuses most of these in its own checks (no self-change, rank ceilings, last-owner guard), but RLS is the boundary and
-- anyone holding a session token can call PostgREST directly.
--
-- ── THE RULE (Taylor, Nov 23) ────────────────────────────────────────────────────────────────────────────────────────
-- For a SIGNED-IN caller (any role that is not trusted, below), on public.crm_org_member:
--   a. Nobody inserts their OWN row, whatever the role (member_owner_guard_self_insert). Recommended over "only not as owner": a
--      real member already has a row (unique (org_id, user_id)), so the only callers who can ever self-insert are non-members
--      holding an admin-tier claim, i.e. platform admins, and joining a firm as admin is an escalation too. No real flow
--      self-inserts through the API: a new firm's first owner is made by provision_org / provision_org_for_owner (below).
--   b. Nobody changes their OWN role, to anything (member_owner_guard_self_role). Wider than "never to owner" on purpose: an admin
--      promoting themselves to senior_admin, or to attorney (0110's close override reads the attorney membership row), is an
--      escalation too, and the app already refuses every self-change ("You can't change your own role.").
--   c. EVERY membership write needs a REAL admin-tier row for the caller in the active firm (owner, admin or senior_admin, read from
--      crm_org_member, not the claim), or the caller is Lectual staff (public.is_platform_admin(), 0050)
--      (member_owner_guard_not_admin). This closes gap 5: a demoted or removed admin's still-valid token no longer writes membership.
--      Lectual staff with no membership row in the firm may write NON-owner members (Taylor, Nov 23), and every such insert, update
--      or delete writes one platform_admin_audit row in the same transaction (see PLATFORM ADMIN AUDIT below).
--   c2. Rank, mirroring the app exactly (see RANK MIRROR below) (member_owner_guard_rank). A caller WITH a membership row in the firm
--      may only give a role at or below their own rank (insert: new.role; update: new.role) and may only act on a member at or below
--      their own rank (update and delete: old.role). The rank is read from the caller's real row, never the claim. Lectual staff with
--      no membership row in the firm are bounded only by the owner rules (d), not by rank (Taylor, Nov 23).
--   d. Inserting an owner row, changing a row to or from owner, or deleting an owner row needs a REAL owner: a crm_org_member row
--      (org_id = the active firm, user_id = auth.uid(), role = 'owner'). That is a table lookup, never current_org_role(), so a
--      platform admin's `owner` claim does not count and a demoted owner's stale token does not count (member_owner_guard_not_owner).
--   e. Only `role` may change on UPDATE: every other column, today's and any added later, is compared as
--      `(to_jsonb(new) - 'role') is distinct from (to_jsonb(old) - 'role')` (0115's form) (member_owner_guard_immutable).
--   f. The firm keeps at least one owner after a signed-in update or delete (member_owner_guard_last_owner). Checked AFTER the
--      statement's rows are written (an AFTER ROW trigger sees the whole statement), under a per-firm advisory lock, so two concurrent
--      transactions each demoting one of two owners cannot both pass, and neither can one statement that removes both.
--   g. A row whose org_id is not the caller's active firm is refused with one fixed sentence before anything else is read
--      (member_owner_guard_wrong_org), so the guard is no oracle about another firm. RLS refuses those writes anyway.
--   h. No session (auth.uid() null) through an untrusted role: member_owner_guard_no_session.
-- All refusals are SQLSTATE 42501 with those stable prefixes. lectual.app's friendlySettingsError
-- (src/app/dashboard/settings/team/errors.ts) already maps any 42501 to "You don't have permission to do that.", so no database
-- text reaches a person and no app change is needed for the refusals themselves.
--
-- DELETE: 0002 has member_delete_admin (above). Deleting an owner row is "touching" one, so it needs a real owner (rule d), and a
-- signed-in delete of the last owner is refused (rule f). Deleting a non-owner row needs a real admin-tier row (rule c). Leaving a
-- firm by deleting your own non-owner row is allowed only for the admin tier (rule c; the app refuses self-removal anyway).

-- ── RANK MIRROR (Taylor, Nov 23: enforce the app's rank limits in the database too) ─────────────────────────────────
-- Mirrored from THIS worktree, not invented. src/lib/auth/roles.ts:
--   export const ROLES = ["owner", "admin", "senior_admin", "attorney", "intake", "paralegal", "law_clerk", "social_media",
--                         "clerk", "viewer"] as const;
--   const RANK: Record<Role, number> = ROLES.reduce((acc, r, i) => ((acc[r] = i), acc), {} as Record<Role, number>);
--   export function hasRole(actual: Role, required: Role): boolean { return RANK[actual] <= RANK[required]; }
-- and the four call sites that bound membership writes:
--   src/lib/settings/members.ts, inviteMember:   if (!hasRole(callerRole, role)) { ... "You can't grant a role above your own"
--   src/lib/settings/members.ts, removeMember:   if (!hasRole(callerRole, current)) { ... "only remove members at or below your own rank"
--   src/lib/settings/admin.ts, setMemberRole:    if (!targetCurrentRole || !hasRole(callerRole, targetCurrentRole)) { ... "only modify
--                                                members at or below your own rank"
--   src/lib/settings/admin.ts, setMemberRole:    if (!hasRole(callerRole, role)) { ... "You can't grant a role above your own"
-- So the database rule is: rank(x) = position of x in that same list (1-based, `array_position`), lower = more privileged, and a
-- write is allowed only when rank(caller) <= rank(new.role) (insert, update) and rank(caller) <= rank(old.role) (update, delete).
-- "At or below" includes EQUAL, as hasRole does: an admin may grant admin, a senior_admin may grant senior_admin but not admin.
-- The only difference from the app: the app's callerRole is the claim (`current_org_role()` RPC); the database uses the caller's real
-- row. A role the list does not know (on either side) is refused (fail closed). The list is written once in the BEFORE guard
-- (v_ranks); the pin test compares it with ROLES from src/lib/auth/roles.ts, so a reorder in the app fails that test.
-- lectual's own copy of src/lib/auth/roles.ts must keep the same order; PRE-APPLY CHECK 7 lists crm_role's labels, and isolation case
-- 21 checks that crm_role's labels and the rank list are the same SET (the order is the app's on purpose, not the enum's).

-- ── PLATFORM ADMIN AUDIT (Taylor, Nov 23) ────────────────────────────────────────────────────────────────────────────
-- 0050's table, verbatim:
--   create table if not exists public.platform_admin_audit (
--     id         uuid primary key default gen_random_uuid(),
--     user_id    uuid not null references auth.users(id) on delete cascade,
--     org_id     uuid not null references public.crm_org(id) on delete cascade,
--     created_at timestamptz not null default now()
--   );
--   RLS enabled with NO policies (members read nothing and write nothing; the service role reads it); one index
--   (user_id, created_at desc); the only writer is public.audit_org_switch(p_org_id), SECURITY DEFINER, which inserts
--   (auth.uid(), p_org_id) only when is_platform_admin(). No event-kind column, no CHECK, no jsonb column. Nothing in either app
--   reads the table.
-- It cannot hold an action, a target or roles, so this file makes the smallest additive change: four NULLABLE columns,
--   action text, target_user_id uuid, old_role public.crm_role, new_role public.crm_role,
-- and one new CHECK, platform_admin_audit_member_write_shape: either all four are null (a workspace entry, as audit_org_switch writes
-- today, so every existing row passes) or action is member_insert (old_role null, new_role set), member_update (both set) or
-- member_delete (old_role set, new_role null), always with target_user_id. No jsonb column exists, so none is added; there was no
-- event-kind CHECK to extend (0101's drop-and-re-add form is not needed). target_user_id has no foreign key on purpose: the record of
-- a write must survive the target's account being deleted. RLS, policies and grants on the table are not changed.
-- The write: an AFTER INSERT OR UPDATE OR DELETE row trigger on crm_org_member (crm_org_member_platform_audit_trg) whose function,
-- public.crm_org_member_platform_audit(), is itself SECURITY DEFINER (search_path = '') and takes NO arguments: it reads only NEW,
-- OLD and TG_OP, so nobody can make it write a row about a firm, target or role they choose. EXECUTE on it is revoked from public,
-- anon, authenticated and service_role (a trigger function needs no EXECUTE to fire). An earlier draft had an argument function in
-- `private` that authenticated could call to write any audit row (a re-review proved a platform admin could forge rows for any firm);
-- it is NOT created by this file. The predicate, in order: auth.uid() null -> nothing; a DELETE of the caller's OWN row (leaving the
-- firm: by AFTER time the row is gone, so the member test would miss them; a re-review found the false member_delete row this
-- skip prevents) -> nothing; pg_trigger_depth() > 1 -> nothing; a DELETE
-- whose firm or auth user is gone (a cascade) -> nothing; the caller has a membership row in that row's firm -> nothing; the caller
-- is not in platform_admin -> nothing; otherwise one row (user_id = auth.uid(), org_id, action, target_user_id, old_role, new_role)
-- built from NEW / OLD. It runs in the same statement, so the audit row and the membership change commit or fail together, and a
-- failing audit insert fails the membership write (nothing catches it). Real members (Lectual staff who also hold a row in that firm
-- included), the service role and cascades write no audit row. No free text is stored. The only other writer of the table is 0050's
-- audit_org_switch(p_org_id), which writes workspace-entry rows only (the four new columns null; it cannot write a member_* row).
--
-- ── WHO IS TRUSTED (passes untouched) ────────────────────────────────────────────────────────────────────────────────
--   * Rows whose parent is gone. Deleting a crm_org, or an auth.users row (Supabase GoTrue, which deletes as supabase_auth_admin),
--     cascades into crm_org_member. The BEFORE trigger sees that delete at pg_trigger_depth() > 1 (0085's and 0107's form) and lets
--     it through. The AFTER ROW trigger does NOT: the cascade's AFTER row events are run at the end of the outer statement, at
--     depth 1, with current_user = the outer role (supabase_auth_admin through GoTrue), which is not trusted. So, for a DELETE by
--     any untrusted role other than authenticated, the last-owner guard asks private.crm_org_member_parent_gone(old.org_id,
--     old.user_id): true when the crm_org row OR the auth.users row no longer exists, and then it passes. authenticated is never
--     asked (it cannot delete a crm_org row or an auth user: no policy, no grant), so it needs no EXECUTE on that helper. Consequence (Taylor's choice): deleting the auth user of a firm's only
--     owner succeeds and leaves the firm with no owner; a signed-in "delete the only owner" still gets last_owner (both parents
--     exist then). No trigger on any table reachable by an API role writes crm_org_member, so depth > 1 is never a way in for a
--     signed-in caller; a future trigger that does must re-check.
--   * current_user = 'service_role' (scripts, the server's admin client, tests' seed helpers).
--   * current_user is the table's OWNER or a member of it (pg_has_role(current_user, relowner, 'MEMBER'); superusers are members of
--     every role). This is what a SECURITY DEFINER function owned by postgres runs as, so provision_org() and
--     provision_org_for_owner() keep creating a new firm's first owner. On Supabase the API roles are NOT members of postgres
--     (postgres is a member of authenticated, not the reverse); PRE-APPLY CHECK 1 confirms it, and that both provisioning
--     functions are owned by the table owner (or a member of it).
-- For that reason the two GUARD functions are SECURITY INVOKER: current_user must be the caller's role. (The audit trigger function is
-- SECURITY DEFINER instead, see above; it has no trust test and keys on auth.uid().) The lookups that have to see past RLS are two
-- SECURITY DEFINER helpers in schema `private`, which PostgREST does not expose (0112's pattern; check `pgrst.db_schemas` does not
-- list it), so no member can call them over rpc or take the owner lock directly. Owned by the table owner, they bypass RLS (the table
-- is not FORCE ROW LEVEL SECURITY), so a guard's read never recurses through a crm_org_member policy.
-- private.crm_org_member_caller_facts(p_lock) answers only for auth.uid() in current_org_id() (the caller's real role, whether they
-- are Lectual staff, the firm's owner count); EXECUTE: authenticated only, because the invoker BEFORE / AFTER guards run as it.
-- private.crm_org_member_parent_gone(org, user) answers only whether two rows exist; EXECUTE: supabase_auth_admin only (narrowed on
-- the re-review): the guard skips it for authenticated, the service role and the table owner are trusted before it is reached, and
-- GoTrue's auth-user delete is the one untrusted cascade. USAGE on `private`: authenticated and supabase_auth_admin.
--
-- WHY A TRIGGER, NOT ONLY TIGHTER POLICIES
--   A policy cannot compare OLD with NEW (self role change, immutables, "touches an owner row"), a policy reading crm_org_member
--   from a crm_org_member policy invites recursion, policies give no stable SQLSTATE prefix, and the last-owner rule needs the whole
--   statement's result. The 0002 policies are left exactly as they are: they still decide WHO may write membership at all (the
--   admin-tier claim) and the triggers only narrow it.
--
-- TABLE REVOKES (decided, Taylor, Nov 23: keep them): this file also revokes TRUNCATE, REFERENCES and TRIGGER (and MAINTAIN on PostgreSQL 17) on
-- crm_org_member from anon and authenticated, as every later tenant table does (0098, 0099, 0103, 0115). TRUNCATE bypasses row
-- triggers and RLS, so without the revoke an API role could still empty the table outside these guards if PostgREST ever exposed it.
-- Nothing in either app needs those privileges. SELECT / INSERT / UPDATE / DELETE are untouched.

-- ── APPLYING IT ──────────────────────────────────────────────────────────────────────────────────────────────────────
-- Additive and idempotent: create schema if not exists, create or replace function, and do $$ blocks that test pg_trigger first
-- (0103's form). Nothing is dropped, no row is changed or deleted, no policy changes; the only grant change is the revoke above; the
-- only schema change outside crm_org_member is four nullable columns and one CHECK on platform_admin_audit (add column if not exists,
-- and a do $$ block that tests pg_constraint). One Supabase MCP apply_migration is safe (no drop, no delete; see 0093 for what hangs
-- it). The audit CHECK is added NOT VALID and then validated in its own statement (0104 / 0105's form). Statement order: the schema,
-- the audit columns and CHECK, the two helpers and their grants, the three trigger functions and their revokes, the three triggers,
-- the table revokes, comments.

-- ── DECISIONS (Taylor, Nov 23) ───────────────────────────────────────────────────────────────────────────────────────
--   D1. Lectual staff with no membership row may invite, re-role and remove NON-owner members; every such write is audited in
--       platform_admin_audit (rule c, PLATFORM ADMIN AUDIT). They never touch an owner row (rule d) and are not bounded by rank.
--   D2. The table revokes of TRUNCATE, REFERENCES, TRIGGER (and MAINTAIN) from anon and authenticated stay in.
--   D3. Rank limits below owner are enforced in the database as well, mirroring the app exactly (rule c2, RANK MIRROR).
--   D4. PRE-APPLY CHECK 4 only shows platform admins' membership rows; nothing removes them.
--   PRODUCTION SHAPE (checked on prod by the director, Nov 23): one platform admin is ALSO the sole real owner member of a firm (owner
--   before becoming staff; the row stays). 0116 treats them as a MEMBER of that firm: every check uses their real owner row (facts reads

--   add a second owner and then leave; as sole owner, deleting their own row is last_owner, and demoting their own row is self_role
--   (rule b runs before the owner and last-owner checks and refuses every self role change; that order is kept on purpose). Leaving

-- ── 2. The lookups that must see past RLS ──────────────────────────────────────────────────────────────────────────
-- Answers for the CALLER in their ACTIVE firm only: their real role there (null = no membership row), whether they are Lectual staff,
-- and the firm's owner count. With p_lock it takes the firm's owner lock first and reads in later statements, so under READ COMMITTED
-- (PostgREST's level) each read sees everything committed before the lock was granted. The lock is held to the end of the
-- transaction. Volatile on purpose: each statement takes a fresh snapshot.
create or replace function private.crm_org_member_caller_facts(
  p_lock boolean,
  out caller_role text,
  out caller_is_platform_admin boolean,
  out owner_count integer
)
language plpgsql
volatile
security definer
set search_path = ''
as $$
declare
  v_org uuid := public.current_org_id();
  v_uid uuid := auth.uid();
begin
  caller_role := null;
  caller_is_platform_admin := false;
  owner_count := 0;
  if v_org is null or v_uid is null then
    return;
  end if;

  if p_lock then
    perform pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended('crm_org_member_owner:' || v_org::text, 0));
  end if;

  select m.role::text into caller_role
    from public.crm_org_member m
   where m.org_id = v_org and m.user_id = v_uid;

  caller_is_platform_admin := coalesce(public.is_platform_admin(), false);

  select count(*)::integer into owner_count
    from public.crm_org_member m
   where m.org_id = v_org and m.role = 'owner';
end;
$$;

revoke all on function private.crm_org_member_caller_facts(boolean) from public;
revoke all on function private.crm_org_member_caller_facts(boolean) from anon;
grant execute on function private.crm_org_member_caller_facts(boolean) to authenticated;

-- True when the membership row's firm OR its auth user no longer exists: the row is being removed by a cascade.

-- ── 3. BEFORE guard: who may write which row ───────────────────────────────────────────────────────────────────────
-- SECURITY INVOKER (no security definer): current_user must be the caller's own role for the trust test.
create or replace function public.crm_org_member_owner_guard()
returns trigger
language plpgsql
set search_path = ''
as $$
declare
  v_uid uuid;
  v_org uuid;
  v_touches_owner boolean := false;
  v_facts record;
  v_caller_rank integer;
  -- The app's ROLES order (src/lib/auth/roles.ts), most privileged first. Change both together.
  v_ranks constant text[] := array['owner', 'admin', 'senior_admin', 'attorney', 'intake', 'paralegal', 'law_clerk', 'social_media', 'clerk', 'viewer'];
begin
  -- 3a. Referential actions (a crm_org or auth.users delete cascading here) pass, as in 0085 and 0107.
  if pg_catalog.pg_trigger_depth() > 1 then
    if tg_op = 'DELETE' then return old; end if;
    return new;
  end if;

  -- 3b. Trusted writers: the service role, and the table's owner (SECURITY DEFINER functions such as provision_org, migrations).
  if current_user = 'service_role'
     or exists (
       select 1 from pg_catalog.pg_class c
        where c.oid = 'public.crm_org_member'::regclass
          and pg_catalog.pg_has_role(current_user, c.relowner, 'MEMBER')
     )
  then
    if tg_op = 'DELETE' then return old; end if;
    return new;
  end if;

  -- 3c. Everyone else must be signed in.
  v_uid := auth.uid();
  if v_uid is null then
    raise exception 'member_owner_guard_no_session: membership changes need a signed-in caller'
      using errcode = '42501';
  end if;

  -- 3d. Only the active firm's rows, refused before anything is read (no oracle about another firm).
  v_org := public.current_org_id();
  if (tg_op in ('INSERT', 'UPDATE') and new.org_id is distinct from v_org)
     or (tg_op in ('UPDATE', 'DELETE') and old.org_id is distinct from v_org)
  then
    raise exception 'member_owner_guard_wrong_org: membership can only change in your active firm'
      using errcode = '42501';
  end if;

  -- 3e. Only the role changes on an update (every other column, today's and any added later).
  if tg_op = 'UPDATE' and (pg_catalog.to_jsonb(new) - 'role') is distinct from (pg_catalog.to_jsonb(old) - 'role') then
    raise exception 'member_owner_guard_immutable: only a member''s role can change'
      using errcode = '42501';
  end if;

  -- 3f. Never your own row: no self-insert (any role), no self role change (any role).
  if tg_op = 'INSERT' and new.user_id = v_uid then
    raise exception 'member_owner_guard_self_insert: you cannot add yourself to a firm'
      using errcode = '42501';
  end if;
  if tg_op = 'UPDATE' and old.user_id = v_uid and new.role is distinct from old.role then
    raise exception 'member_owner_guard_self_role: you cannot change your own role'
      using errcode = '42501';
  end if;

  -- 3g. Does this write create, change or remove an owner row?
  if tg_op = 'INSERT' then
    v_touches_owner := new.role = 'owner';
  elsif tg_op = 'UPDATE' then
    v_touches_owner := new.role is distinct from old.role and (old.role = 'owner' or new.role = 'owner');
  else
    v_touches_owner := old.role = 'owner';
  end if;

  -- 3h. The caller's REAL standing in this firm (the table, not the claim). The owner lock is taken only when an owner row is touched.
  select f.caller_role, f.caller_is_platform_admin, f.owner_count into v_facts
    from private.crm_org_member_caller_facts(v_touches_owner) f;

  if not (coalesce(v_facts.caller_role, '') in ('owner', 'admin', 'senior_admin') or coalesce(v_facts.caller_is_platform_admin, false)) then
    raise exception 'member_owner_guard_not_admin: only a current owner or admin of this firm can change its members'
      using errcode = '42501';
  end if;

  if v_touches_owner and coalesce(v_facts.caller_role, '') <> 'owner' then
    raise exception 'member_owner_guard_not_owner: only an owner of this firm can add, change or remove an owner'
      using errcode = '42501';
  end if;

  -- 3i. Rank (the app's hasRole): a caller with a membership row gives and acts on roles at or below their own only. Lectual staff
  --     with no membership row here are bounded by the owner rules above, not by rank.
  if v_facts.caller_role is not null then
    v_caller_rank := pg_catalog.array_position(v_ranks, v_facts.caller_role);
    if v_caller_rank is null
       or (tg_op in ('INSERT', 'UPDATE') and coalesce(pg_catalog.array_position(v_ranks, new.role::text), 0) < v_caller_rank)
       or (tg_op in ('UPDATE', 'DELETE') and coalesce(pg_catalog.array_position(v_ranks, old.role::text), 0) < v_caller_rank)
    then
      raise exception 'member_owner_guard_rank: you can only give or change roles at or below your own'
        using errcode = '42501';
    end if;
  end if;

  if tg_op = 'DELETE' then return old; end if;
  return new;
end;
$$;


-- ── 4. AFTER guard: the firm keeps at least one owner ──────────────────────────────────────────────────────────────
-- AFTER ROW triggers fire once the statement has written every row, so the count includes the whole statement (two owners removed in
-- one statement) and, after the lock, every transaction committed before it (two owners demoted in two transactions).
create or replace function public.crm_org_member_last_owner_guard()
returns trigger
language plpgsql
set search_path = ''
as $$
declare
  v_facts record;
begin
  if pg_catalog.pg_trigger_depth() > 1 then
    return null;
  end if;

  if current_user = 'service_role'
     or exists (
       select 1 from pg_catalog.pg_class c
        where c.oid = 'public.crm_org_member'::regclass
          and pg_catalog.pg_has_role(current_user, c.relowner, 'MEMBER')
     )
  then
    return null;
  end if;

  -- 4a. A cascade from a deleted auth user (GoTrue, as supabase_auth_admin): its AFTER events run at depth 1 as the outer role, so
  --     test the parents. authenticated cannot delete a firm or an auth user, so it is never asked (and has no EXECUTE on it). The
  --     two IFs are nested on purpose: PL/pgSQL plans an expression (and checks EXECUTE on every function in it) the first time it
  --     runs, so the helper must sit in an expression authenticated never reaches.
  if tg_op = 'DELETE' and current_user <> 'authenticated' then
    if private.crm_org_member_parent_gone(old.org_id, old.user_id) then
      return null;
    end if;
  end if;

  -- Only a row that WAS an owner and no longer is one can leave the firm without an owner.
  if old.role is distinct from 'owner' then
    return null;
  end if;
  if tg_op = 'UPDATE' and new.role = 'owner' then
    return null;
  end if;

  if old.org_id is distinct from public.current_org_id() then
    raise exception 'member_owner_guard_wrong_org: membership can only change in your active firm'
      using errcode = '42501';
  end if;

  select f.owner_count into v_facts from private.crm_org_member_caller_facts(true) f;
  if coalesce(v_facts.owner_count, 0) < 1 then
    raise exception 'member_owner_guard_last_owner: a firm must keep at least one owner'
      using errcode = '42501';
  end if;

  return null;
end;
$$;

revoke execute on function public.crm_org_member_owner_guard() from public;
revoke execute on function public.crm_org_member_owner_guard() from anon;
revoke execute on function public.crm_org_member_owner_guard() from authenticated;
revoke execute on function public.crm_org_member_last_owner_guard() from public;
revoke execute on function public.crm_org_member_last_owner_guard() from anon;
revoke execute on function public.crm_org_member_last_owner_guard() from authenticated;


-- ── 4b. AFTER audit: Lectual staff's membership writes ─────────────────────────────────────────────────────────────
-- SECURITY DEFINER, no arguments: reads only NEW, OLD and TG_OP, so no caller can choose what it records. Nothing catches an error.
create or replace function public.crm_org_member_platform_audit()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_uid uuid := auth.uid();
  v_org uuid;
  v_target uuid;
begin
  if v_uid is null then
    return null;
  end if;

  -- Leaving a firm: the caller's OWN row is already gone when this AFTER trigger runs, so the member test below would miss them.
  -- A member deleting their own row is never a platform-admin write. (A self-update cannot reach here: self_role refuses it.)
  if tg_op = 'DELETE' and old.user_id = v_uid then
    return null;
  end if;

  if pg_catalog.pg_trigger_depth() > 1 then
    return null;
  end if;

  if tg_op = 'DELETE' then
    v_org := old.org_id;
    v_target := old.user_id;
  else
    v_org := new.org_id;
    v_target := new.user_id;
  end if;

  -- A cascade from a deleted firm or auth user.
  if tg_op = 'DELETE'
     and (not exists (select 1 from public.crm_org o where o.id = v_org)
          or not exists (select 1 from auth.users u where u.id = v_target))
  then
    return null;
  end if;

  -- A member of this firm (Lectual staff who also hold a row here included) is not audited.
  if exists (select 1 from public.crm_org_member m where m.org_id = v_org and m.user_id = v_uid) then
    return null;
  end if;

  if not exists (select 1 from public.platform_admin p where p.user_id = v_uid) then
    return null;
  end if;

  if tg_op = 'INSERT' then
    insert into public.platform_admin_audit (user_id, org_id, action, target_user_id, old_role, new_role)
    values (v_uid, v_org, 'member_insert', v_target, null, new.role);
  elsif tg_op = 'UPDATE' then
    insert into public.platform_admin_audit (user_id, org_id, action, target_user_id, old_role, new_role)
    values (v_uid, v_org, 'member_update', v_target, old.role, new.role);
  else
    insert into public.platform_admin_audit (user_id, org_id, action, target_user_id, old_role, new_role)
    values (v_uid, v_org, 'member_delete', v_target, old.role, null);
  end if;

  return null;
end;
$$;

revoke execute on function public.crm_org_member_platform_audit() from public;
revoke execute on function public.crm_org_member_platform_audit() from anon;
revoke execute on function public.crm_org_member_platform_audit() from authenticated;
revoke execute on function public.crm_org_member_platform_audit() from service_role;
